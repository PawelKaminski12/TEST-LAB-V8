// V8 DUAL ENGINE — PANEL PRIORITY PACK V5
// Nadpisania do pełnego skryptu V4.
// Cel: hierarchia 1H/4H/1D, tłumienie szumu 1H i wybór najważniejszego alarmu do PANELU.
// Nie zmienia logiki LONG/Tactical. Brak wykonywania transakcji.

function confluencePriorityForTf_(count, tf) {
  if (tf === '1D') {
    if (count >= 2) return 'KRYTYCZNY';
    return 'MOCNY';
  }
  if (count >= 3) return 'KRYTYCZNY';
  if (count === 2) return 'MOCNY';
  return 'OSTRZEŻENIE';
}

function buildExtremeAlarms_(ss, tacticalJson) {
  const cfg = getExtremeSettings_(ss);
  if (!cfg.enabled) return [];
  const alarms = [];
  (tacticalJson.assets || []).forEach(a => {
    const t = a.timeframes || {};
    ['1H','4H','1D'].forEach(tf => {
      const x = t[tf] || {};
      const rsi = Number(x.rsi14), mfi = Number(x.mfi14), fomo = Number(x.fomo_score_0_10);
      const macdPct = Number(x.macd_hist_percentile), macdZ = Number(x.macd_hist_zscore);
      const ts = x.last_close_time_utc || '';
      const signals = [];
      if (Number.isFinite(rsi) && rsi >= cfg.rsiHigh) signals.push(rsi >= 80 ? 'RSI SKRAJNE WYKUPIENIE' : 'RSI WYKUPIENIE');
      if (Number.isFinite(rsi) && rsi <= cfg.rsiLow) signals.push(rsi <= 20 ? 'RSI SKRAJNE WYPRZEDANIE' : 'RSI WYPRZEDANIE');
      if (Number.isFinite(mfi)) {
        if (mfi >= cfg.mfiExtremeHigh) signals.push('MFI SKRAJNE WYKUPIENIE');
        else if (mfi >= cfg.mfiWarnHigh) signals.push('MFI WYKUPIENIE');
        if (mfi <= cfg.mfiExtremeLow) signals.push('MFI SKRAJNE WYPRZEDANIE');
        else if (mfi <= cfg.mfiWarnLow) signals.push('MFI WYPRZEDANIE');
      }
      const ml = String(x.macd_extreme_label || 'NORMAL');
      if (ml === 'EXTREME_POSITIVE') signals.push('MACD SKRAJNIE DODATNIE');
      else if (ml === 'EXTREME_NEGATIVE') signals.push('MACD SKRAJNIE UJEMNE');
      else if (ml === 'WARNING_POSITIVE') signals.push('MACD MOCNO DODATNIE');
      else if (ml === 'WARNING_NEGATIVE') signals.push('MACD MOCNO UJEMNE');
      if (Number.isFinite(fomo) && fomo >= cfg.fomoHard) signals.push(fomo >= 9 ? 'FOMO SKRAJNE' : 'FOMO HARD');
      if (!signals.length) return;
      const count = signals.length;
      const priority = confluencePriorityForTf_(count, tf);
      const tfWeight = timeframeWeight_(tf);
      const reason = confluenceReason_(a.symbol, tf, count, priority, tfWeight, signals);
      alarms.push(makeConfluenceAlarm_(a.symbol, tf, signals, rsi, mfi, macdPct, macdZ, fomo, priority, tfWeight, reason, ts));
    });
  });
  return alarms.sort(compareExtremePriority_);
}

function compareExtremePriority_(a,b) {
  const p = {KRYTYCZNY:3,MOCNY:2,OSTRZEŻENIE:1};
  const t = {'1D':3,'4H':2,'1H':1};
  return (p[b.priority]-p[a.priority]) || (t[b.tf]-t[a.tf]) || (b.signalCount-a.signalCount) || String(a.symbol).localeCompare(String(b.symbol));
}

function panelEligibleExtreme_(a) {
  // Pojedyncze ostrzeżenie 1H pozostaje w ALARMY_EKSTREMOW i HISTORII,
  // ale nie zaśmieca głównego PANELU.
  return !(a.tf === '1H' && a.priority === 'OSTRZEŻENIE' && Number(a.signalCount) === 1);
}

function selectMainPanelExtreme_(alarms) {
  const eligible = (alarms || []).filter(panelEligibleExtreme_).slice().sort(compareExtremePriority_);
  return eligible.length ? eligible[0] : null;
}

function updateMainPanelExtremeSummary_(sheet, alarms) {
  if (!sheet || sheet.getLastRow() < 2) return;
  const bySymbol = {};
  (alarms || []).forEach(a => {
    if (!panelEligibleExtreme_(a)) return;
    (bySymbol[a.symbol] = bySymbol[a.symbol] || []).push(a);
  });
  const values = sheet.getDataRange().getValues();
  const head = values[0].map(String);
  const symIdx = head.indexOf('AKTYWO');
  const alertIdx = head.indexOf('ALERT');
  if (symIdx < 0 || alertIdx < 0) return;
  for (let i=1; i<values.length; i++) {
    const sym = String(values[i][symIdx] || '').trim();
    if (!sym) continue;
    const best = selectMainPanelExtreme_(bySymbol[sym] || []);
    let base = String(values[i][alertIdx] || '').trim();
    base = base.replace(/\s*\|\s*[🟡🟠🔴]\s*(OSTRZEŻENIE|MOCNY|KRYTYCZNY)\s+(1H|4H|1D)\s+—\s+\d+\s+sygnał(?:y|ów)?/g,'').trim();
    if (best) {
      const icon = best.priority === 'KRYTYCZNY' ? '🔴' : best.priority === 'MOCNY' ? '🟠' : '🟡';
      const n = best.signalCount === 1 ? '1 sygnał' : best.signalCount < 5 ? best.signalCount + ' sygnały' : best.signalCount + ' sygnałów';
      const extra = icon+' '+best.priority+' '+best.tf+' — '+n;
      values[i][alertIdx] = (!base || base === 'BRAK') ? extra : base+' | '+extra;
    } else {
      values[i][alertIdx] = base || 'BRAK';
    }
  }
  sheet.getRange(1,1,values.length,values[0].length).setValues(values);
  formatPanel_(sheet);
  formatMainPanelAlerts_(sheet);
}

function notifyNewExtremeAlarms_(ss, oldKeys, alarms) {
  const fresh = (alarms || []).filter(a => !oldKeys[a.key]);
  const actionable = fresh.filter(a => !(a.tf === '1H' && a.priority === 'OSTRZEŻENIE' && Number(a.signalCount) === 1));
  if (!actionable.length) return;
  const text = actionable.map(a => a.symbol+' '+a.tf+' — '+a.priority+' — '+a.signalCount+' sygnał(y): '+a.type).join('\n');
  ss.toast(text.slice(0,250), 'NOWE EKSTREMA V8', 8);
  const cfg = getExtremeSettings_(ss);
  if (cfg.emailOn && cfg.emailTo) MailApp.sendEmail(cfg.emailTo, 'V8 DUAL ENGINE — alarm ekstremum', text);
}

function formatMainPanelAlerts_(sheet) {
  if (!sheet || sheet.getLastRow() < 2) return;
  const range = sheet.getRange(2,7,sheet.getLastRow()-1,1);
  const rules = (sheet.getConditionalFormatRules() || []).filter(r => {
    const rs = r.getRanges();
    return !rs.some(x => x.getColumn() === 7 && x.getNumColumns() === 1);
  });
  rules.push(SpreadsheetApp.newConditionalFormatRule().whenTextContains('🔴 KRYTYCZNY').setBackground('#f4cccc').setFontColor('#990000').setBold(true).setRanges([range]).build());
  rules.push(SpreadsheetApp.newConditionalFormatRule().whenTextContains('🟠 MOCNY').setBackground('#fce5cd').setFontColor('#b45f06').setBold(true).setRanges([range]).build());
  rules.push(SpreadsheetApp.newConditionalFormatRule().whenTextContains('🟡 OSTRZEŻENIE').setBackground('#fff2cc').setFontColor('#7f6000').setRanges([range]).build());
  sheet.setConditionalFormatRules(rules);
}
