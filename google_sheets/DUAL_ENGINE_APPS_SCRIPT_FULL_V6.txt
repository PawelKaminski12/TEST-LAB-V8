function V8_SETUP() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  Object.values(V8_SHEETS).forEach(name => getOrCreateSheet_(ss, name));
  setupPortfolioLong_(ss.getSheetByName(V8_SHEETS.portfolioLong));
  setupPortfolioTactical_(ss.getSheetByName(V8_SHEETS.portfolioTactical));
  setupSettings_(ss.getSheetByName(V8_SHEETS.settings));
  setupExtremes_(ss.getSheetByName(V8_SHEETS.extremes));
  formatPanel_(ss.getSheetByName(V8_SHEETS.panel));
  ss.toast('Układ V8 DUAL ENGINE gotowy', 'V8', 4);
}

function ODSWIEZ_WSZYSTKO() {
  const lock = LockService.getDocumentLock();
  if (!lock.tryLock(1000)) throw new Error('Odświeżanie już trwa.');
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    V8_SETUP();

    const oldPanel = readPanelState_(ss.getSheetByName(V8_SHEETS.panel));
    const oldAlarms = readActiveExtremeKeys_(ss.getSheetByName(V8_SHEETS.extremes));

    const panelCsv = fetchText_(V8_FILES.panel);
    const machineCsv = fetchText_(V8_FILES.machine);
    const longJson = JSON.parse(fetchText_(V8_FILES.long));
    const tacticalJson = JSON.parse(fetchText_(V8_FILES.tactical));
    const readinessJson = JSON.parse(fetchText_(V8_FILES.tacticalReadiness));
    const phaseJson = JSON.parse(fetchText_(V8_FILES.marketPhase));
    const snapshotJson = JSON.parse(fetchText_(V8_FILES.snapshot));
    const labJson = JSON.parse(fetchText_(V8_FILES.labReport));
    const etfStatusJson = JSON.parse(fetchText_(V8_FILES.etfStatus));
    const etfHistoryCsv = fetchText_(V8_FILES.etfHistoryCsv);
    const etfConfirmationJson = JSON.parse(fetchText_(V8_FILES.etfConfirmation));
    const mrCsv = fetchText_(V8_FILES.marketReader);

    writePolishPanel_(ss.getSheetByName(V8_SHEETS.panel), panelCsv);
    writeLongEngine_(ss.getSheetByName(V8_SHEETS.long), longJson);
    writeTacticalEngine_(ss.getSheetByName(V8_SHEETS.tactical), tacticalJson, readinessJson, phaseJson);
    writeFlowEngine_(ss.getSheetByName(V8_SHEETS.flow), phaseJson, snapshotJson);
    writeEtfPanel_(ss.getSheetByName(V8_SHEETS.etf), etfStatusJson, etfConfirmationJson);
    writeEtfHistory_(ss.getSheetByName(V8_SHEETS.etfHistory), etfHistoryCsv);
    writeLabPanel_(ss.getSheetByName(V8_SHEETS.lab), labJson);
    writeCsvSheet_(ss.getSheetByName(V8_SHEETS.marketReader), mrCsv, false);

    const machineSheet = getOrCreateSheet_(ss, 'DANE_MASZYNOWNIA');
    writeCsvSheet_(machineSheet, machineCsv, false);
    machineSheet.hideSheet();

    const alarms = buildExtremeAlarms_(ss, tacticalJson);
    writeExtremeAlarms_(ss.getSheetByName(V8_SHEETS.extremes), alarms);
    updateMainPanelExtremeSummary_(ss.getSheetByName(V8_SHEETS.panel), alarms);
    appendExtremeHistory_(ss, oldAlarms, alarms);
    notifyNewExtremeAlarms_(ss, oldAlarms, alarms);

    formatPanel_(ss.getSheetByName(V8_SHEETS.panel));
    appendHistoryOnChanges_(ss, oldPanel, readPanelState_(ss.getSheetByName(V8_SHEETS.panel)));
    stampRefresh_(ss);

    const active = alarms.filter(a => a.aktywny).length;
    ss.toast('LONG + TACTICAL + ETF + LAB odświeżone. Aktywne ekstrema: ' + active, 'V8 DUAL ENGINE', 5);
  } finally {
    lock.releaseLock();
  }
}

function SPRAWDZ_EKSTREMA() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  V8_SETUP();
  const oldAlarms = readActiveExtremeKeys_(ss.getSheetByName(V8_SHEETS.extremes));
  const tacticalJson = JSON.parse(fetchText_(V8_FILES.tactical));
  const alarms = buildExtremeAlarms_(ss, tacticalJson);
  writeExtremeAlarms_(ss.getSheetByName(V8_SHEETS.extremes), alarms);
  updateMainPanelExtremeSummary_(ss.getSheetByName(V8_SHEETS.panel), alarms);
  appendExtremeHistory_(ss, oldAlarms, alarms);
  notifyNewExtremeAlarms_(ss, oldAlarms, alarms);
  ss.toast('Sprawdzono ekstrema 1H / 4H / 1D', 'V8', 4);
}

function fetchText_(url) {
  const response = UrlFetchApp.fetch(url, {
    muteHttpExceptions: true,
    followRedirects: true,
    headers: {'Cache-Control': 'no-cache'}
  });
  const code = response.getResponseCode();
  if (code < 200 || code >= 300) throw new Error('HTTP ' + code + ' dla ' + url);
  return response.getContentText('UTF-8');
}

function getOrCreateSheet_(ss, name) {
  return ss.getSheetByName(name) || ss.insertSheet(name);
}

function writeCsvSheet_(sheet, csvText, preserveCompactPanel) {
  const rows = Utilities.parseCsv(csvText);
  sheet.clearContents();
  if (!rows.length) return;
  sheet.getRange(1, 1, rows.length, rows[0].length).setValues(rows);
  sheet.setFrozenRows(1);
  if (preserveCompactPanel) {
    const maxCols = 7;
    if (sheet.getMaxColumns() > maxCols) sheet.hideColumns(maxCols + 1, sheet.getMaxColumns() - maxCols);
  }
}

function writePolishPanel_(sheet, csvText) {
  const rows = Utilities.parseCsv(csvText);
  sheet.clearContents();
  if (!rows.length) return;
  const head = rows[0].map(String);
  const idx = {}; head.forEach((h,i) => idx[h] = i);
  const out = [['AKTYWO','CENA','LONG','TACTICAL','RYZYKO 0–10','ZGODNOŚĆ','ALERT']];
  rows.slice(1).forEach(r => {
    if (!String(r[idx['AKTYWO']] || '').trim()) return;
    out.push([
      r[idx['AKTYWO']], r[idx['CENA']], translateLong_(r[idx['LONG']]),
      translateTactical_(r[idx['TACTICAL']]), r[idx['RYZYKO_0_10']],
      translateAgreement_(r[idx['ZGODNOSC']]), translateAlert_(r[idx['ALERT']])
    ]);
  });
  sheet.getRange(1,1,out.length,out[0].length).setValues(out);
  sheet.setFrozenRows(1);
}

  const headers = ['AKTYWO','JAKOŚĆ','STATUS DANYCH','STREFY','TREND','FOMO','ROTACJA','FAZA RYNKU','FLOW LONG 0–10','FLOW OCENA','ETF 5D','ETF 20D','ETF 30D','DECYZJA','DCA1','DCA2','DCA3','BLOKADY'];

    a.trend_score_0_4, a.fomo_score_0_10, translateGeneric_(a.rotation_label), translateMarketPhase_(a.market_phase),

function writeEtfHistory_(sheet, csvText) {
  const rows = Utilities.parseCsv(csvText);
  sheet.clearContents();
  if (!rows.length) return;
  const keep = rows.length > 121 ? [rows[0]].concat(rows.slice(-120)) : rows;
  sheet.getRange(1,1,keep.length,keep[0].length).setValues(keep);
  sheet.setFrozenRows(1);
  sheet.getRange(1,1,1,keep[0].length).setFontWeight('bold');
  if (keep.length > 1) sheet.getRange(2,2,keep.length-1,3).setNumberFormat('0.00');
  sheet.getRange(1,1,keep.length,keep[0].length).setWrap(true);
  sheet.autoResizeColumns(1,keep[0].length);
}

function labTfText_(x) {
  if (!x || x.status !== 'OK') return 'BRAK PEŁNYCH DANYCH';
  return 'TREND '+fmtCell_(x.trend_0_4)+'/4 | RSI '+fmtCell_(x.rsi14)+' | MFI '+fmtCell_(x.mfi14)+' | FOMO '+fmtCell_(x.fomo_0_10)+'/10';
}

function fmtCell_(v) {
  if (v === null || v === undefined || v === '') return 'brak danych';
  const n = Number(v); return Number.isFinite(n) ? Math.round(n*100)/100 : String(v);
}
function valueOrBlank_(v) { const n=Number(v); return Number.isFinite(n) ? n : ''; }

function etfMeaning_(five, twenty, accel, fallback) {
  five=Number(five); twenty=Number(twenty); accel=Number(accel);
  if (Number.isFinite(five) && Number.isFinite(twenty)) {
    if (five>0 && twenty>0) return Number.isFinite(accel) && accel>0 ? 'NAPŁYW PRZYSPIESZA — zainteresowanie rośnie' : 'NAPŁYW KAPITAŁU — zainteresowanie dodatnie';
    if (five<0 && twenty<0) return Number.isFinite(accel) && accel<0 ? 'ODPŁYW PRZYSPIESZA — zainteresowanie słabnie' : 'ODPŁYW KAPITAŁU — zainteresowanie słabnie';
    return 'PRZEPŁYWY MIESZANE — brak jednolitego kierunku';
  }
  return fallback ? translateEtfTrend_(fallback) : 'ZA MAŁO DANYCH';
}

function globalEtfMeaning_(i) {
  const b5=Number(i.btc_etf_5d_usdm), b20=Number(i.btc_etf_20d_usdm), e5=Number(i.eth_etf_5d_usdm), e20=Number(i.eth_etf_20d_usdm);
  if ([b5,b20,e5,e20].every(Number.isFinite)) {

function buildExtremeAlarms_(ss, tacticalJson) {
  const cfg = getExtremeSettings_(ss);
  if (!cfg.enabled) return [];

  const alarms = [];
  (tacticalJson.assets || []).forEach(a => {
    const t = a.timeframes || {};
    ['1H','4H','1D'].forEach(tf => {
      const x = t[tf] || {};
      const rsi = Number(x.rsi14);
      const mfi = Number(x.mfi14);
      const fomo = Number(x.fomo_score_0_10);
      const macdPct = Number(x.macd_hist_percentile);
      const macdZ = Number(x.macd_hist_zscore);
      const ts = x.last_close_time_utc || '';
      const signals = [];

      if (Number.isFinite(rsi) && rsi >= cfg.rsiHigh) signals.push(rsi >= 80 ? 'RSI SKRAJNE WYKUPIENIE' : 'RSI WYKUPIENIE');
      if (Number.isFinite(rsi) && rsi <= cfg.rsiLow) signals.push(rsi <= 20 ? 'RSI SKRAJNE WYPRZEDANIE' : 'RSI WYPRZEDANIE');

      if (Number.isFinite(mfi)) {
        if (mfi >= cfg.mfiExtremeHigh) signals.push('MFI SKRAJNE WYKUPIENIE');
        else if (mfi >= cfg.mfiWarnHigh) signals.push('MFI WYKUPIENIE');
        if (mfi <= cfg.mfiExtremeLow) signals.push('MFI SKRAJNE WYPRZEDANIE');
        else if (mfi <= cfg.mfiWarnLow) signals.push('MFI WYPRZEDANIE');
      }

      const ml = String(x.macd_extreme_label || 'NORMAL');
      if (ml === 'EXTREME_POSITIVE') signals.push('MACD SKRAJNIE DODATNIE');
      else if (ml === 'EXTREME_NEGATIVE') signals.push('MACD SKRAJNIE UJEMNE');
      else if (ml === 'WARNING_POSITIVE') signals.push('MACD MOCNO DODATNIE');
      else if (ml === 'WARNING_NEGATIVE') signals.push('MACD MOCNO UJEMNE');

      if (Number.isFinite(fomo) && fomo >= cfg.fomoHard) signals.push(fomo >= 9 ? 'FOMO SKRAJNE' : 'FOMO HARD');

      if (!signals.length) return;

      const count = signals.length;
      const priority = confluencePriority_(count);
      const tfWeight = timeframeWeight_(tf);
      const reason = confluenceReason_(a.symbol, tf, count, priority, tfWeight, signals);
      alarms.push(makeConfluenceAlarm_(a.symbol, tf, signals, rsi, mfi, macdPct, macdZ, fomo, priority, tfWeight, reason, ts));
    });
  });

  return alarms.sort((a,b) => {
    const p = {KRYTYCZNY:3,MOCNY:2,OSTRZEŻENIE:1};
    const w = {'NAJWYŻSZA':3,'WYSOKA':2,'NISKA':1};
    return (p[b.priority]-p[a.priority]) || (w[b.tfWeight]-w[a.tfWeight]) || String(a.symbol).localeCompare(String(b.symbol));
  });
}

function confluencePriority_(count) {
  if (count >= 3) return 'KRYTYCZNY';
  if (count === 2) return 'MOCNY';
  return 'OSTRZEŻENIE';
}

function timeframeWeight_(tf) {
  if (tf === '1D') return 'NAJWYŻSZA';
  if (tf === '4H') return 'WYSOKA';
  return 'NISKA';
}

function confluenceReason_(symbol, tf, count, priority, tfWeight, signals) {
  const n = count === 1 ? '1 sygnał' : count < 5 ? count + ' sygnały' : count + ' sygnałów';
  return symbol + ' ' + tf + ': ' + n + ' zgodne. Priorytet ' + priority + ', waga interwału ' + tfWeight + '. ' + signals.join(' + ') + '.';
}

    macdPct:Number.isFinite(macdPct)?macdPct:'',

    fomo:Number.isFinite(fomo)?fomo:'',
    reason:reason,

  };
}

// V8 MAIN PANEL ALERT SUMMARY V4
// Nie zmienia liczby kolumn PANELU. Używa istniejącej kolumny ALERT.
// Pokazuje najwyższy aktywny alarm ekstremum dla danego aktywa.

function updateMainPanelExtremeSummary_(sheet, alarms) {
  if (!sheet || sheet.getLastRow() < 2) return;
  const values = sheet.getDataRange().getValues();
  const head = values[0].map(String);
  const symCol = head.indexOf('AKTYWO');
  const alertCol = head.indexOf('ALERT');
  if (symCol < 0 || alertCol < 0) return;

  const best = {};
  const priorityRank = {KRYTYCZNY:3, MOCNY:2, OSTRZEŻENIE:1};
  const tfRank = {'1D':3, '4H':2, '1H':1};

  (alarms || []).forEach(a => {
    const sym = String(a.symbol || '').trim();
    if (!sym) return;
    const candidate = {
      priority: a.priority || a.severity || 'OSTRZEŻENIE',
      tf: a.tf || '',
      count: Number(a.signalCount || 1),
      type: a.type || ''
    };
    const old = best[sym];
    if (!old ||
        (priorityRank[candidate.priority] || 0) > (priorityRank[old.priority] || 0) ||
        ((priorityRank[candidate.priority] || 0) === (priorityRank[old.priority] || 0) && (tfRank[candidate.tf] || 0) > (tfRank[old.tf] || 0))) {
      best[sym] = candidate;
    }
  });

  const out = [];
  for (let i = 1; i < values.length; i++) {
    const sym = String(values[i][symCol] || '').trim();
    const current = String(values[i][alertCol] || '').trim();
    const a = best[sym];
    if (!a) {
      out.push([current || 'BRAK']);
      continue;
    }
    const prefix = a.priority === 'KRYTYCZNY' ? '🔴' : a.priority === 'MOCNY' ? '🟠' : '🟡';
    const extreme = prefix + ' ' + a.priority + ' ' + a.tf + ' — ' + a.count + (a.count === 1 ? ' sygnał' : a.count < 5 ? ' sygnały' : ' sygnałów');
    out.push([current && current !== 'BRAK' ? current + ' | ' + extreme : extreme]);
  }

  sheet.getRange(2, alertCol + 1, out.length, 1).setValues(out);
  formatMainPanelAlertColumn_(sheet, alertCol + 1);
}

function formatMainPanelAlertColumn_(sheet, alertCol1Based) {
  if (!sheet || sheet.getLastRow() < 2) return;
  const range = sheet.getRange(2, alertCol1Based, sheet.getLastRow() - 1, 1);
  const vals = range.getDisplayValues();
  range.setFontWeight('normal');
  for (let i = 0; i < vals.length; i++) {
    const v = String(vals[i][0] || '');
    const cell = sheet.getRange(i + 2, alertCol1Based);
    if (v.indexOf('🔴 KRYTYCZNY') >= 0) cell.setFontWeight('bold').setBackground('#f4cccc');
    else if (v.indexOf('🟠 MOCNY') >= 0) cell.setFontWeight('bold').setBackground('#fce5cd');
    else if (v.indexOf('🟡 OSTRZEŻENIE') >= 0) cell.setBackground('#fff2cc');
    else cell.setBackground(null);
  }
}

// Nadpisania do pełnego skryptu V4.
// Cel: hierarchia 1H/4H/1D, tłumienie szumu 1H i wybór najważniejszego alarmu do PANELU.
// Nie zmienia logiki LONG/Tactical. Brak wykonywania transakcji.

function confluencePriorityForTf_(count, tf) {
  if (tf === '1D') {
    if (count >= 2) return 'KRYTYCZNY';
    return 'MOCNY';
  }
  if (count >= 3) return 'KRYTYCZNY';
  if (count === 2) return 'MOCNY';
  return 'OSTRZEŻENIE';
}

function buildExtremeAlarms_(ss, tacticalJson) {
  const cfg = getExtremeSettings_(ss);
  if (!cfg.enabled) return [];
  const alarms = [];
  (tacticalJson.assets || []).forEach(a => {
    const t = a.timeframes || {};
    ['1H','4H','1D'].forEach(tf => {
      const x = t[tf] || {};
      const rsi = Number(x.rsi14), mfi = Number(x.mfi14), fomo = Number(x.fomo_score_0_10);
      const macdPct = Number(x.macd_hist_percentile), macdZ = Number(x.macd_hist_zscore);
      const ts = x.last_close_time_utc || '';
      const signals = [];
      if (Number.isFinite(rsi) && rsi >= cfg.rsiHigh) signals.push(rsi >= 80 ? 'RSI SKRAJNE WYKUPIENIE' : 'RSI WYKUPIENIE');
      if (Number.isFinite(rsi) && rsi <= cfg.rsiLow) signals.push(rsi <= 20 ? 'RSI SKRAJNE WYPRZEDANIE' : 'RSI WYPRZEDANIE');
      if (Number.isFinite(mfi)) {
        if (mfi >= cfg.mfiExtremeHigh) signals.push('MFI SKRAJNE WYKUPIENIE');
        else if (mfi >= cfg.mfiWarnHigh) signals.push('MFI WYKUPIENIE');
        if (mfi <= cfg.mfiExtremeLow) signals.push('MFI SKRAJNE WYPRZEDANIE');
        else if (mfi <= cfg.mfiWarnLow) signals.push('MFI WYPRZEDANIE');
      }
      const ml = String(x.macd_extreme_label || 'NORMAL');
      if (ml === 'EXTREME_POSITIVE') signals.push('MACD SKRAJNIE DODATNIE');
      else if (ml === 'EXTREME_NEGATIVE') signals.push('MACD SKRAJNIE UJEMNE');
      else if (ml === 'WARNING_POSITIVE') signals.push('MACD MOCNO DODATNIE');
      else if (ml === 'WARNING_NEGATIVE') signals.push('MACD MOCNO UJEMNE');
      if (Number.isFinite(fomo) && fomo >= cfg.fomoHard) signals.push(fomo >= 9 ? 'FOMO SKRAJNE' : 'FOMO HARD');
      if (!signals.length) return;
      const count = signals.length;
      const priority = confluencePriorityForTf_(count, tf);
      const tfWeight = timeframeWeight_(tf);
      const reason = confluenceReason_(a.symbol, tf, count, priority, tfWeight, signals);
      alarms.push(makeConfluenceAlarm_(a.symbol, tf, signals, rsi, mfi, macdPct, macdZ, fomo, priority, tfWeight, reason, ts));
    });
  });
  return alarms.sort(compareExtremePriority_);
}

function compareExtremePriority_(a,b) {
  const p = {KRYTYCZNY:3,MOCNY:2,OSTRZEŻENIE:1};
  const t = {'1D':3,'4H':2,'1H':1};
  return (p[b.priority]-p[a.priority]) || (t[b.tf]-t[a.tf]) || (b.signalCount-a.signalCount) || String(a.symbol).localeCompare(String(b.symbol));
}

  if (a.aktywny !== b.aktywny) return a.aktywny ? -1 : 1;
  return compareExtremePriority_(a,b);
}

function writeExtremeAlarms_(sheet, alarms) {
  const headers = ['AKTYWNY','AKTYWO','INTERWAŁ','ŚWIEŻOŚĆ','WIEK H','LIMIT H','ZBIEŻNOŚĆ','PRIORYTET','WAGA INTERWAŁU','SYGNAŁY','RSI','MFI','MACD %','MACD Z','FOMO','OPIS','CZAS DANYCH'];
  sheet.clearContents();
  sheet.getRange(1,1,1,headers.length).setValues([headers]);
  const rows = (alarms || []).map(a => [
    a.aktywny ? 'TAK' : 'NIE',a.symbol,a.tf,a.freshness,a.ageHours,a.limitHours,a.signalCount,a.priority,a.tfWeight,a.type,
    a.rsi,a.mfi,a.macdPct,a.macdZ,a.fomo,a.reason,a.ts
  ]);
  if (rows.length) sheet.getRange(2,1,rows.length,headers.length).setValues(rows);
  else sheet.getRange(2,1,1,headers.length).setValues([['NIE','','','BRAK','', '',0,'BRAK','BRAK','BRAK AKTYWNYCH EKSTREMÓW','','','','','','','']]);
  sheet.setFrozenRows(1);
  sheet.setFrozenColumns(4);
  sheet.getRange(1,1,1,headers.length).setFontWeight('bold');
  sheet.getRange(1,1,Math.max(2,rows.length+1),headers.length).setWrap(true).setVerticalAlignment('middle');
  [72,78,75,95,70,70,80,100,105,300,70,70,80,80,70,420,175].forEach((w,i)=>sheet.setColumnWidth(i+1,w));
  formatFreshnessAlarmSheet_(sheet, rows.length);
}

function formatFreshnessAlarmSheet_(sheet, rowCount) {
  const n = Math.max(1,rowCount);
  const freshRange = sheet.getRange(2,4,n,1);
  const activeRange = sheet.getRange(2,1,n,1);
  const priorityRange = sheet.getRange(2,8,n,1);
  const rules = (sheet.getConditionalFormatRules() || []).filter(r => {
    return !r.getRanges().some(x => x.getSheet().getName() === sheet.getName());
  });
  rules.push(SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('STARY').setBackground('#eeeeee').setFontColor('#777777').setRanges([freshRange]).build());
  rules.push(SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('BRAK CZASU').setBackground('#d9d2e9').setFontColor('#674ea7').setRanges([freshRange]).build());
  rules.push(SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('AKTUALNY').setBackground('#d9ead3').setFontColor('#274e13').setRanges([freshRange]).build());
  rules.push(SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('NIE').setBackground('#eeeeee').setFontColor('#777777').setRanges([activeRange]).build());
  rules.push(SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('KRYTYCZNY').setBackground('#f4cccc').setFontColor('#990000').setBold(true).setRanges([priorityRange]).build());
  rules.push(SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('MOCNY').setBackground('#fce5cd').setFontColor('#b45f06').setBold(true).setRanges([priorityRange]).build());
  rules.push(SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('OSTRZEŻENIE').setBackground('#fff2cc').setFontColor('#7f6000').setRanges([priorityRange]).build());
  sheet.setConditionalFormatRules(rules);
}

function readActiveExtremeKeys_(sheet) {
  const out = {};
  if (!sheet || sheet.getLastRow() < 2) return out;
  const rows = sheet.getRange(2,1,sheet.getLastRow()-1,Math.min(17,sheet.getLastColumn())).getValues();
  rows.forEach(r => {
    if (String(r[0]).toUpperCase() !== 'TAK') return;
    out[[r[1],r[2],r[9]].join('|')] = true;
  });
  return out;
}

function panelEligibleExtreme_(a) {
  if (!a || !a.aktywny || a.freshness !== 'AKTUALNY') return false;
  return !(a.tf === '1H' && a.priority === 'OSTRZEŻENIE' && Number(a.signalCount) === 1);
}

function appendExtremeHistory_(ss, oldKeys, alarms) {
  const sh = ss.getSheetByName(V8_SHEETS.history);
  if (sh.getLastRow() === 0) {
    sh.appendRow(['CZAS','AKTYWO','LONG','TACTICAL','RYZYKO','ZGODNOŚĆ','ALERT','POWÓD']);
    sh.setFrozenRows(1);
  }
  const now = new Date();
  (alarms || []).forEach(a => {
    if (a.aktywny && !oldKeys[a.key]) {
      sh.appendRow([now,a.symbol,'','',a.priority,'','NOWE EKSTREMUM '+a.tf,a.reason]);
    } else if (!a.aktywny && oldKeys[a.key]) {
      sh.appendRow([now,a.symbol,'','','','','ALARM WYGASŁ '+a.tf,'Dane przestały być świeże: '+a.reason]);
    }
  });
}

function notifyNewExtremeAlarms_(ss, oldKeys, alarms) {
  const fresh = (alarms || []).filter(a => a.aktywny && a.freshness === 'AKTUALNY' && !oldKeys[a.key]);
  const actionable = fresh.filter(panelEligibleExtreme_);
  if (!actionable.length) return;
  const text = actionable.map(a => a.symbol+' '+a.tf+' — '+a.priority+' — '+a.signalCount+' sygnał(y), dane '+a.ageHours+'h: '+a.type).join('\n');
  ss.toast(text.slice(0,250), 'NOWE AKTUALNE EKSTREMA V8', 8);
  const cfg = getExtremeSettings_(ss);
  if (cfg.emailOn && cfg.emailTo) MailApp.sendEmail(cfg.emailTo, 'V8 DUAL ENGINE — aktualny alarm ekstremum', text);
}

function updateMainPanelExtremeSummary_(sheet, alarms) {
  if (!sheet || sheet.getLastRow() < 2) return;
  const bySymbol = {};
  (alarms || []).forEach(a => {
    if (!panelEligibleExtreme_(a)) return;
    (bySymbol[a.symbol] = bySymbol[a.symbol] || []).push(a);
  });
  const values = sheet.getDataRange().getValues();
  const head = values[0].map(String);
  const symIdx = head.indexOf('AKTYWO');
  const alertIdx = head.indexOf('ALERT');
  if (symIdx < 0 || alertIdx < 0) return;
  for (let i=1; i<values.length; i++) {
    const sym = String(values[i][symIdx] || '').trim();
    if (!sym) continue;
    const best = selectMainPanelExtreme_(bySymbol[sym] || []);
    let base = String(values[i][alertIdx] || '').trim();
    base = base.replace(/\s*\|\s*[🟡🟠🔴]\s*(OSTRZEŻENIE|MOCNY|KRYTYCZNY)\s+(1H|4H|1D)\s+—\s+\d+\s+sygnał(?:y|ów)?(?:\s+\[[^\]]+\])?/g,'').trim();
    if (best) {
      const icon = best.priority === 'KRYTYCZNY' ? '🔴' : best.priority === 'MOCNY' ? '🟠' : '🟡';
      const n = best.signalCount === 1 ? '1 sygnał' : best.signalCount < 5 ? best.signalCount+' sygnały' : best.signalCount+' sygnałów';
      const age = best.ageHours === '' ? '' : ' ['+best.ageHours+'h]';
      const extra = icon+' '+best.priority+' '+best.tf+' — '+n+age;
      values[i][alertIdx] = (!base || base === 'BRAK') ? extra : base+' | '+extra;
    } else {
      values[i][alertIdx] = base || 'BRAK';
    }
  }
  sheet.getRange(1,1,values.length,values[0].length).setValues(values);
  formatPanel_(sheet);
  formatMainPanelAlerts_(sheet);
}
