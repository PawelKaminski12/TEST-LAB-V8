function V8_SETUP() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  Object.values(V8_SHEETS).forEach(name => getOrCreateSheet_(ss, name));
  setupPortfolioLong_(ss.getSheetByName(V8_SHEETS.portfolioLong));
  setupPortfolioTactical_(ss.getSheetByName(V8_SHEETS.portfolioTactical));
  setupSettings_(ss.getSheetByName(V8_SHEETS.settings));
  setupExtremes_(ss.getSheetByName(V8_SHEETS.extremes));
  formatPanel_(ss.getSheetByName(V8_SHEETS.panel));
  ss.toast('Układ V8 DUAL ENGINE gotowy', 'V8', 4);
}

function ODSWIEZ_WSZYSTKO() {
  const lock = LockService.getDocumentLock();
  if (!lock.tryLock(1000)) throw new Error('Odświeżanie już trwa.');
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    V8_SETUP();

    const oldPanel = readPanelState_(ss.getSheetByName(V8_SHEETS.panel));
    const oldAlarms = readActiveExtremeKeys_(ss.getSheetByName(V8_SHEETS.extremes));

    const panelCsv = fetchText_(V8_FILES.panel);
    const machineCsv = fetchText_(V8_FILES.machine);
    const longJson = JSON.parse(fetchText_(V8_FILES.long));
    const tacticalJson = JSON.parse(fetchText_(V8_FILES.tactical));
    const readinessJson = JSON.parse(fetchText_(V8_FILES.tacticalReadiness));
    const phaseJson = JSON.parse(fetchText_(V8_FILES.marketPhase));
    const snapshotJson = JSON.parse(fetchText_(V8_FILES.snapshot));
    const labJson = JSON.parse(fetchText_(V8_FILES.labReport));
    const etfStatusJson = JSON.parse(fetchText_(V8_FILES.etfStatus));
    const etfHistoryCsv = fetchText_(V8_FILES.etfHistoryCsv);
    const etfConfirmationJson = JSON.parse(fetchText_(V8_FILES.etfConfirmation));
    const mrCsv = fetchText_(V8_FILES.marketReader);

    writePolishPanel_(ss.getSheetByName(V8_SHEETS.panel), panelCsv);
    writeLongEngine_(ss.getSheetByName(V8_SHEETS.long), longJson);
    writeTacticalEngine_(ss.getSheetByName(V8_SHEETS.tactical), tacticalJson, readinessJson, phaseJson);
    writeFlowEngine_(ss.getSheetByName(V8_SHEETS.flow), phaseJson, snapshotJson);
    writeEtfPanel_(ss.getSheetByName(V8_SHEETS.etf), etfStatusJson, etfConfirmationJson);
    writeEtfHistory_(ss.getSheetByName(V8_SHEETS.etfHistory), etfHistoryCsv);
    writeLabPanel_(ss.getSheetByName(V8_SHEETS.lab), labJson);
    writeCsvSheet_(ss.getSheetByName(V8_SHEETS.marketReader), mrCsv, false);

    const machineSheet = getOrCreateSheet_(ss, 'DANE_MASZYNOWNIA');
    writeCsvSheet_(machineSheet, machineCsv, false);
    machineSheet.hideSheet();

    const alarms = buildExtremeAlarms_(ss, tacticalJson);
    writeExtremeAlarms_(ss.getSheetByName(V8_SHEETS.extremes), alarms);
    updateMainPanelExtremeSummary_(ss.getSheetByName(V8_SHEETS.panel), alarms);
    appendExtremeHistory_(ss, oldAlarms, alarms);
    notifyNewExtremeAlarms_(ss, oldAlarms, alarms);

    formatPanel_(ss.getSheetByName(V8_SHEETS.panel));
    applyUnifiedThemeV10_(ss);
    appendHistoryOnChanges_(ss, oldPanel, readPanelState_(ss.getSheetByName(V8_SHEETS.panel)));
    stampRefresh_(ss);
    applyUnifiedV8Theme_(ss);

    const active = alarms.filter(a => a.aktywny).length;
    ss.toast('LONG + TACTICAL + ETF + LAB odświeżone. Aktywne ekstrema: ' + active, 'V8 DUAL ENGINE', 5);
  } finally {
    lock.releaseLock();
  }
}

function SPRAWDZ_EKSTREMA() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  V8_SETUP();
  const oldAlarms = readActiveExtremeKeys_(ss.getSheetByName(V8_SHEETS.extremes));
  const tacticalJson = JSON.parse(fetchText_(V8_FILES.tactical));
  const alarms = buildExtremeAlarms_(ss, tacticalJson);
  writeExtremeAlarms_(ss.getSheetByName(V8_SHEETS.extremes), alarms);
  updateMainPanelExtremeSummary_(ss.getSheetByName(V8_SHEETS.panel), alarms);
  appendExtremeHistory_(ss, oldAlarms, alarms);
  notifyNewExtremeAlarms_(ss, oldAlarms, alarms);
  ss.toast('Sprawdzono ekstrema 1H / 4H / 1D', 'V8', 4);
}

function fetchText_(url) {
  const response = UrlFetchApp.fetch(url, {
    muteHttpExceptions: true,
    followRedirects: true,
    headers: {'Cache-Control': 'no-cache'}
  });
  const code = response.getResponseCode();
  if (code < 200 || code >= 300) throw new Error('HTTP ' + code + ' dla ' + url);
  return response.getContentText('UTF-8');
}

function getOrCreateSheet_(ss, name) {
  return ss.getSheetByName(name) || ss.insertSheet(name);
}

function writeCsvSheet_(sheet, csvText, preserveCompactPanel) {
  const rows = Utilities.parseCsv(csvText);
  sheet.clearContents();
  if (!rows.length) return;
  sheet.getRange(1, 1, rows.length, rows[0].length).setValues(rows);
  sheet.setFrozenRows(1);
  if (preserveCompactPanel) {
    const maxCols = 7;
    if (sheet.getMaxColumns() > maxCols) sheet.hideColumns(maxCols + 1, sheet.getMaxColumns() - maxCols);
  }
}

function writePolishPanel_(sheet, csvText) {
  const rows = Utilities.parseCsv(csvText);
  sheet.clearContents();
  if (!rows.length) return;
  const head = rows[0].map(String);
  const idx = {}; head.forEach((h,i) => idx[h] = i);
  const out = [['AKTYWO','CENA','LONG','TACTICAL','RYZYKO 0–10','ZGODNOŚĆ','ALERT']];
  rows.slice(1).forEach(r => {
    if (!String(r[idx['AKTYWO']] || '').trim()) return;
    out.push([
      r[idx['AKTYWO']], r[idx['CENA']], translateLong_(r[idx['LONG']]),
      translateTactical_(r[idx['TACTICAL']]), r[idx['RYZYKO_0_10']],
      translateAgreement_(r[idx['ZGODNOSC']]), translateAlert_(r[idx['ALERT']])
    ]);
  });
  sheet.getRange(1,1,out.length,out[0].length).setValues(out);
  sheet.setFrozenRows(1);
}

function writeLongEngine_(sheet, data) {
  const headers = ['AKTYWO','JAKOŚĆ','STATUS DANYCH','STREFY','TREND','FOMO','ROTACJA','FAZA RYNKU','FLOW LONG 0–10','FLOW OCENA','ETF 5D','ETF 20D','ETF 30D','DECYZJA','DCA1','DCA2','DCA3','BLOKADY'];

    a.trend_score_0_4, a.fomo_score_0_10, translateGeneric_(a.rotation_label), translateMarketPhase_(a.market_phase),

    const p20=c.price_vs_etf_20d || {};

function writeEtfHistory_(sheet, csvText) {
  const rows = Utilities.parseCsv(csvText);
  sheet.clearContents();
  if (!rows.length) return;
  const keep = rows.length > 121 ? [rows[0]].concat(rows.slice(-120)) : rows;
  sheet.getRange(1,1,keep.length,keep[0].length).setValues(keep);
  sheet.setFrozenRows(1);
  sheet.getRange(1,1,1,keep[0].length).setFontWeight('bold');
  if (keep.length > 1) sheet.getRange(2,2,keep.length-1,3).setNumberFormat('0.00');
  sheet.getRange(1,1,keep.length,keep[0].length).setWrap(true);
  sheet.autoResizeColumns(1,keep[0].length);
}

function labTfText_(x) {
  if (!x || x.status !== 'OK') return 'BRAK PEŁNYCH DANYCH';
  return 'TREND '+fmtCell_(x.trend_0_4)+'/4 | RSI '+fmtCell_(x.rsi14)+' | MFI '+fmtCell_(x.mfi14)+' | FOMO '+fmtCell_(x.fomo_0_10)+'/10';
}

function fmtCell_(v) {
  if (v === null || v === undefined || v === '') return 'brak danych';
  const n = Number(v); return Number.isFinite(n) ? Math.round(n*100)/100 : String(v);
}
function valueOrBlank_(v) { const n=Number(v); return Number.isFinite(n) ? n : ''; }

function etfMeaning_(five, twenty, accel, fallback) {
  five=Number(five); twenty=Number(twenty); accel=Number(accel);
  if (Number.isFinite(five) && Number.isFinite(twenty)) {
    if (five>0 && twenty>0) return Number.isFinite(accel) && accel>0 ? 'NAPŁYW PRZYSPIESZA — zainteresowanie rośnie' : 'NAPŁYW KAPITAŁU — zainteresowanie dodatnie';
    if (five<0 && twenty<0) return Number.isFinite(accel) && accel<0 ? 'ODPŁYW PRZYSPIESZA — zainteresowanie słabnie' : 'ODPŁYW KAPITAŁU — zainteresowanie słabnie';
    return 'PRZEPŁYWY MIESZANE — brak jednolitego kierunku';
  }
  return fallback ? translateEtfTrend_(fallback) : 'ZA MAŁO DANYCH';
}

function globalEtfMeaning_(i) {
  const b5=Number(i.btc_etf_5d_usdm), b20=Number(i.btc_etf_20d_usdm), e5=Number(i.eth_etf_5d_usdm), e20=Number(i.eth_etf_20d_usdm);
  if ([b5,b20,e5,e20].every(Number.isFinite)) {
    if (b5>0 && b20>0 && e5>0 && e20>0) return 'KAPITAŁ NAPŁYWA RÓWNOCZEŚNIE DO BTC I ETH — szerokie zainteresowanie instytucjonalne krypto rośnie.';

function buildExtremeAlarms_(ss, tacticalJson) {
  const cfg = getExtremeSettings_(ss);
  if (!cfg.enabled) return [];

  const alarms = [];
  (tacticalJson.assets || []).forEach(a => {
    const t = a.timeframes || {};
    ['1H','4H','1D'].forEach(tf => {
      const x = t[tf] || {};
      const rsi = Number(x.rsi14);
      const mfi = Number(x.mfi14);
      const fomo = Number(x.fomo_score_0_10);
      const macdPct = Number(x.macd_hist_percentile);
      const macdZ = Number(x.macd_hist_zscore);
      const ts = x.last_close_time_utc || '';
      const signals = [];

      if (Number.isFinite(rsi) && rsi >= cfg.rsiHigh) signals.push(rsi >= 80 ? 'RSI SKRAJNE WYKUPIENIE' : 'RSI WYKUPIENIE');
      if (Number.isFinite(rsi) && rsi <= cfg.rsiLow) signals.push(rsi <= 20 ? 'RSI SKRAJNE WYPRZEDANIE' : 'RSI WYPRZEDANIE');

      if (Number.isFinite(mfi)) {
        if (mfi >= cfg.mfiExtremeHigh) signals.push('MFI SKRAJNE WYKUPIENIE');
        else if (mfi >= cfg.mfiWarnHigh) signals.push('MFI WYKUPIENIE');
        if (mfi <= cfg.mfiExtremeLow) signals.push('MFI SKRAJNE WYPRZEDANIE');
        else if (mfi <= cfg.mfiWarnLow) signals.push('MFI WYPRZEDANIE');
      }

      const ml = String(x.macd_extreme_label || 'NORMAL');
      if (ml === 'EXTREME_POSITIVE') signals.push('MACD SKRAJNIE DODATNIE');
      else if (ml === 'EXTREME_NEGATIVE') signals.push('MACD SKRAJNIE UJEMNE');
      else if (ml === 'WARNING_POSITIVE') signals.push('MACD MOCNO DODATNIE');
      else if (ml === 'WARNING_NEGATIVE') signals.push('MACD MOCNO UJEMNE');

      if (Number.isFinite(fomo) && fomo >= cfg.fomoHard) signals.push(fomo >= 9 ? 'FOMO SKRAJNE' : 'FOMO HARD');

      if (!signals.length) return;

      const count = signals.length;
      const priority = confluencePriority_(count);
      const tfWeight = timeframeWeight_(tf);
      const reason = confluenceReason_(a.symbol, tf, count, priority, tfWeight, signals);
      alarms.push(makeConfluenceAlarm_(a.symbol, tf, signals, rsi, mfi, macdPct, macdZ, fomo, priority, tfWeight, reason, ts));
    });
  });

  return alarms.sort((a,b) => {
    const p = {KRYTYCZNY:3,MOCNY:2,OSTRZEŻENIE:1};
    const w = {'NAJWYŻSZA':3,'WYSOKA':2,'NISKA':1};
    return (p[b.priority]-p[a.priority]) || (w[b.tfWeight]-w[a.tfWeight]) || String(a.symbol).localeCompare(String(b.symbol));
  });
}

function confluencePriority_(count) {
  if (count >= 3) return 'KRYTYCZNY';
  if (count === 2) return 'MOCNY';
  return 'OSTRZEŻENIE';
}

function timeframeWeight_(tf) {
  if (tf === '1D') return 'NAJWYŻSZA';
  if (tf === '4H') return 'WYSOKA';
  return 'NISKA';
}

function confluenceReason_(symbol, tf, count, priority, tfWeight, signals) {
  const n = count === 1 ? '1 sygnał' : count < 5 ? count + ' sygnały' : count + ' sygnałów';
  return symbol + ' ' + tf + ': ' + n + ' zgodne. Priorytet ' + priority + ', waga interwału ' + tfWeight + '. ' + signals.join(' + ') + '.';
}

    fomo:Number.isFinite(fomo)?fomo:'',

}

// V8 MAIN PANEL ALERT SUMMARY V4
// Nie zmienia liczby kolumn PANELU. Używa istniejącej kolumny ALERT.
// Pokazuje najwyższy aktywny alarm ekstremum dla danego aktywa.

function updateMainPanelExtremeSummary_(sheet, alarms) {
  if (!sheet || sheet.getLastRow() < 2) return;
  const values = sheet.getDataRange().getValues();
  const head = values[0].map(String);
  const symCol = head.indexOf('AKTYWO');
  const alertCol = head.indexOf('ALERT');
  if (symCol < 0 || alertCol < 0) return;

  const best = {};
  const priorityRank = {KRYTYCZNY:3, MOCNY:2, OSTRZEŻENIE:1};
  const tfRank = {'1D':3, '4H':2, '1H':1};

  (alarms || []).forEach(a => {
    const sym = String(a.symbol || '').trim();
    if (!sym) return;
    const candidate = {
      priority: a.priority || a.severity || 'OSTRZEŻENIE',
      tf: a.tf || '',
      count: Number(a.signalCount || 1),
      type: a.type || ''
    };
    const old = best[sym];
    if (!old ||
        (priorityRank[candidate.priority] || 0) > (priorityRank[old.priority] || 0) ||
        ((priorityRank[candidate.priority] || 0) === (priorityRank[old.priority] || 0) && (tfRank[candidate.tf] || 0) > (tfRank[old.tf] || 0))) {
      best[sym] = candidate;
    }
  });

  const out = [];
  for (let i = 1; i < values.length; i++) {
    const sym = String(values[i][symCol] || '').trim();
    const current = String(values[i][alertCol] || '').trim();
    const a = best[sym];
    if (!a) {
      out.push([current || 'BRAK']);
      continue;
    }
    const prefix = a.priority === 'KRYTYCZNY' ? '🔴' : a.priority === 'MOCNY' ? '🟠' : '🟡';
    const extreme = prefix + ' ' + a.priority + ' ' + a.tf + ' — ' + a.count + (a.count === 1 ? ' sygnał' : a.count < 5 ? ' sygnały' : ' sygnałów');
    out.push([current && current !== 'BRAK' ? current + ' | ' + extreme : extreme]);
  }

  sheet.getRange(2, alertCol + 1, out.length, 1).setValues(out);
  formatMainPanelAlertColumn_(sheet, alertCol + 1);
}

function formatMainPanelAlertColumn_(sheet, alertCol1Based) {
  if (!sheet || sheet.getLastRow() < 2) return;
  const range = sheet.getRange(2, alertCol1Based, sheet.getLastRow() - 1, 1);
  const vals = range.getDisplayValues();
  range.setFontWeight('normal');
  for (let i = 0; i < vals.length; i++) {
    const v = String(vals[i][0] || '');
    const cell = sheet.getRange(i + 2, alertCol1Based);
    if (v.indexOf('🔴 KRYTYCZNY') >= 0) cell.setFontWeight('bold').setBackground('#f4cccc');
    else if (v.indexOf('🟠 MOCNY') >= 0) cell.setFontWeight('bold').setBackground('#fce5cd');
    else if (v.indexOf('🟡 OSTRZEŻENIE') >= 0) cell.setBackground('#fff2cc');
    else cell.setBackground(null);
  }
}

// Nadpisania do pełnego skryptu V4.
// Cel: hierarchia 1H/4H/1D, tłumienie szumu 1H i wybór najważniejszego alarmu do PANELU.
// Nie zmienia logiki LONG/Tactical. Brak wykonywania transakcji.

function confluencePriorityForTf_(count, tf) {
  if (tf === '1D') {
    if (count >= 2) return 'KRYTYCZNY';
    return 'MOCNY';
  }
  if (count >= 3) return 'KRYTYCZNY';
  if (count === 2) return 'MOCNY';
  return 'OSTRZEŻENIE';
}

function buildExtremeAlarms_(ss, tacticalJson) {
  const cfg = getExtremeSettings_(ss);
  if (!cfg.enabled) return [];
  const alarms = [];
  (tacticalJson.assets || []).forEach(a => {
    const t = a.timeframes || {};
    ['1H','4H','1D'].forEach(tf => {
      const x = t[tf] || {};
      const rsi = Number(x.rsi14), mfi = Number(x.mfi14), fomo = Number(x.fomo_score_0_10);
      const macdPct = Number(x.macd_hist_percentile), macdZ = Number(x.macd_hist_zscore);
      const ts = x.last_close_time_utc || '';
      const signals = [];
      if (Number.isFinite(rsi) && rsi >= cfg.rsiHigh) signals.push(rsi >= 80 ? 'RSI SKRAJNE WYKUPIENIE' : 'RSI WYKUPIENIE');
      if (Number.isFinite(rsi) && rsi <= cfg.rsiLow) signals.push(rsi <= 20 ? 'RSI SKRAJNE WYPRZEDANIE' : 'RSI WYPRZEDANIE');
      if (Number.isFinite(mfi)) {
        if (mfi >= cfg.mfiExtremeHigh) signals.push('MFI SKRAJNE WYKUPIENIE');
        else if (mfi >= cfg.mfiWarnHigh) signals.push('MFI WYKUPIENIE');
        if (mfi <= cfg.mfiExtremeLow) signals.push('MFI SKRAJNE WYPRZEDANIE');
        else if (mfi <= cfg.mfiWarnLow) signals.push('MFI WYPRZEDANIE');
      }
      const ml = String(x.macd_extreme_label || 'NORMAL');
      if (ml === 'EXTREME_POSITIVE') signals.push('MACD SKRAJNIE DODATNIE');
      else if (ml === 'EXTREME_NEGATIVE') signals.push('MACD SKRAJNIE UJEMNE');
      else if (ml === 'WARNING_POSITIVE') signals.push('MACD MOCNO DODATNIE');
      else if (ml === 'WARNING_NEGATIVE') signals.push('MACD MOCNO UJEMNE');
      if (Number.isFinite(fomo) && fomo >= cfg.fomoHard) signals.push(fomo >= 9 ? 'FOMO SKRAJNE' : 'FOMO HARD');
      if (!signals.length) return;
      const count = signals.length;
      const priority = confluencePriorityForTf_(count, tf);
      const tfWeight = timeframeWeight_(tf);
      const reason = confluenceReason_(a.symbol, tf, count, priority, tfWeight, signals);
      alarms.push(makeConfluenceAlarm_(a.symbol, tf, signals, rsi, mfi, macdPct, macdZ, fomo, priority, tfWeight, reason, ts));
    });
  });
  return alarms.sort(compareExtremePriority_);
}

function compareExtremePriority_(a,b) {
  const p = {KRYTYCZNY:3,MOCNY:2,OSTRZEŻENIE:1};
  const t = {'1D':3,'4H':2,'1H':1};
  return (p[b.priority]-p[a.priority]) || (t[b.tf]-t[a.tf]) || (b.signalCount-a.signalCount) || String(a.symbol).localeCompare(String(b.symbol));
}

// V8 DUAL ENGINE — MULTI-TF CONFLUENCE V7
// Dodaje zgodność międzyinterwałową 1H / 4H / 1D dla świeżych alarmów.
// Nie zmienia logiki LONG ani TACTICAL. Nie wykonuje transakcji.

function alarmDirection_(a) {
  const signals = a.signals || [];
  let hot = 0, cold = 0;
  signals.forEach(s => {
    const x = String(s || '');
    if (x.indexOf('WYKUPIENIE') >= 0 || x.indexOf('DODATNIE') >= 0 || x.indexOf('FOMO') >= 0) hot++;
    if (x.indexOf('WYPRZEDANIE') >= 0 || x.indexOf('UJEMNE') >= 0) cold++;
  });
  if (hot > 0 && cold === 0) return 'PRZEGRZANIE';
  if (cold > 0 && hot === 0) return 'WYPRZEDANIE';
  return 'MIESZANE';
}

function buildMultiTfConfluence_(alarms) {
  const bySymbol = {};
  (alarms || []).filter(a => a.aktywny).forEach(a => {
    const dir = alarmDirection_(a);
    if (dir === 'MIESZANE') return;
    const key = a.symbol + '|' + dir;
    (bySymbol[key] = bySymbol[key] || []).push(a);
  });

  const out = [];
  Object.keys(bySymbol).forEach(key => {
    const group = bySymbol[key].slice().sort((a,b) => ({'1D':3,'4H':2,'1H':1}[b.tf]||0)-({'1D':3,'4H':2,'1H':1}[a.tf]||0));
    const tfSet = {};
    group.forEach(a => tfSet[a.tf] = a);
    const tfs = ['1H','4H','1D'].filter(tf => tfSet[tf]);
    if (tfs.length < 2) return;

    const [symbol, direction] = key.split('|');
    const has4h1d = !!tfSet['4H'] && !!tfSet['1D'];
    const all3 = tfs.length === 3;
    const priority = all3 || has4h1d ? 'KRYTYCZNY' : 'MOCNY';
    const totalSignals = group.reduce((s,a) => s + Number(a.signalCount || 0), 0);
    const type = 'ZGODNOŚĆ WIELU INTERWAŁÓW — ' + direction + ' — ' + tfs.join('+');
    const reason = symbol + ': ' + direction + ' potwierdzone na ' + tfs.join(' + ') + '. Łącznie ' + totalSignals + ' sygnałów. ' + (has4h1d ? 'Potwierdzenie obejmuje 4H + 1D.' : 'Potwierdzenie obejmuje dwa interwały.');
    const newestTs = group.map(a => Date.parse(String(a.ts||''))).filter(Number.isFinite).sort((a,b)=>b-a)[0];

    out.push({
      aktywny:true,
      symbol:symbol,
      tf:'WIELE TF',
      multiTf:true,
      multiTfList:tfs,
      direction:direction,
      type:type,
      signalCount:totalSignals,
      priority:priority,
      severity:priority,
      tfWeight:'WIELOINTERWAŁOWA',
      signals:group.map(a => a.tf+': '+a.type),
      rsi:'',mfi:'',macdPct:'',macdZ:'',fomo:'',
      reason:reason,
      ts:Number.isFinite(newestTs) ? new Date(newestTs).toISOString() : '',
      freshness:'AKTUALNY',ageHours:'',limitHours:'',
      key:[symbol,'WIELE_TF',direction,tfs.join('+')].join('|')
    });
  });
  return out;
}

function buildExtremeAlarms_(ss, tacticalJson) {
  const cfg = getExtremeSettings_(ss);

  const base = [];
  (tacticalJson.assets || []).forEach(a => {
    const t = a.timeframes || {};
    ['1H','4H','1D'].forEach(tf => {
      const x = t[tf] || {};

      const macdPct = Number(x.macd_hist_percentile), macdZ = Number(x.macd_hist_zscore);
      const ts = x.last_close_time_utc || '';
      const signals = [];
      if (Number.isFinite(rsi) && rsi >= cfg.rsiHigh) signals.push(rsi >= 80 ? 'RSI SKRAJNE WYKUPIENIE' : 'RSI WYKUPIENIE');
      if (Number.isFinite(rsi) && rsi <= cfg.rsiLow) signals.push(rsi <= 20 ? 'RSI SKRAJNE WYPRZEDANIE' : 'RSI WYPRZEDANIE');
      if (Number.isFinite(mfi)) {
        if (mfi >= cfg.mfiExtremeHigh) signals.push('MFI SKRAJNE WYKUPIENIE');
        else if (mfi >= cfg.mfiWarnHigh) signals.push('MFI WYKUPIENIE');
        if (mfi <= cfg.mfiExtremeLow) signals.push('MFI SKRAJNE WYPRZEDANIE');
        else if (mfi <= cfg.mfiWarnLow) signals.push('MFI WYPRZEDANIE');
      }
      const ml = String(x.macd_extreme_label || 'NORMAL');
      if (ml === 'EXTREME_POSITIVE') signals.push('MACD SKRAJNIE DODATNIE');
      else if (ml === 'EXTREME_NEGATIVE') signals.push('MACD SKRAJNIE UJEMNE');
      else if (ml === 'WARNING_POSITIVE') signals.push('MACD MOCNO DODATNIE');
      else if (ml === 'WARNING_NEGATIVE') signals.push('MACD MOCNO UJEMNE');

    });

// V8 DUAL ENGINE — ALERT STATE TRANSITIONS V8
// Warstwa dynamiki alarmów: NOWY / NASILA SIĘ / BEZ ZMIAN / SŁABNIE / WYGASŁ.
// Nie zmienia logiki LONG/Tactical. Nie wykonuje transakcji.

function alarmDirectionV8_(a) {
  if (a && a.direction) return String(a.direction);
  return classifyAlarmDirection_(String((a && a.type) || ''));
}

function alarmStateIdV8_(a) {
  return [String(a.symbol||''), String(a.tf||''), alarmDirectionV8_(a)].join('|');
}

function alarmStrengthV8_(a) {
  const p = {OSTRZEŻENIE:1,MOCNY:2,KRYTYCZNY:3};
  const t = {'1H':1,'4H':2,'1D':3,'WIELE TF':4};
  const tfCount = a && a.multiTfList ? a.multiTfList.length : 1;
  return (p[String(a.priority||'')]||0)*100 + (t[String(a.tf||'')]||0)*10 + tfCount*3 + Number(a.signalCount||0);
}

function readExtremeStateMapV8_(sheet) {
  const out = {};
  if (!sheet || sheet.getLastRow() < 2) return out;
  const values = sheet.getDataRange().getValues();
  const h = values[0].map(String);
  const idx = {}; h.forEach((x,i)=>idx[x]=i);
  if (idx['AKTYWO'] == null || idx['INTERWAŁ'] == null) return out;
  values.slice(1).forEach(r => {
    const sym = String(r[idx['AKTYWO']]||'').trim();
    const tf = String(r[idx['INTERWAŁ']]||'').trim();
    if (!sym || !tf) return;
    const direction = idx['KIERUNEK'] == null ? 'MIESZANY' : String(r[idx['KIERUNEK']]||'MIESZANY');
    const a = {
      symbol:sym, tf:tf, direction:direction,
      aktywny:String(r[idx['AKTYWNY']]||'').toUpperCase()==='TAK',
      priority:idx['PRIORYTET']==null?'':String(r[idx['PRIORYTET']]||''),
      signalCount:idx['ZBIEŻNOŚĆ']==null?0:Number(r[idx['ZBIEŻNOŚĆ']]||0),
      state:idx['STAN']==null?'':String(r[idx['STAN']]||''),
      type:idx['SYGNAŁY']==null?'':String(r[idx['SYGNAŁY']]||''),
      ts:idx['CZAS DANYCH']==null?'':r[idx['CZAS DANYCH']]
    };
    out[[sym,tf,direction].join('|')] = a;
  });
  return out;
}

function determineAlarmStateV8_(prev, cur) {
  if (!cur.aktywny) return prev && prev.aktywny ? 'WYGASŁ' : 'BEZ ZMIAN';
  if (!prev || !prev.aktywny) return 'NOWY';
  const before = alarmStrengthV8_(prev), now = alarmStrengthV8_(cur);
  if (now > before) return 'NASILA SIĘ';
  if (now < before) return 'SŁABNIE';
  return 'BEZ ZMIAN';
}

function applyAlarmStateTransitionsV8_(previous, alarms) {
  const seen = {};
  (alarms||[]).forEach(a => {
    a.direction = alarmDirectionV8_(a);
    a.stateId = alarmStateIdV8_(a);
    const prev = previous[a.stateId];
    a.previousPriority = prev ? prev.priority : '';
    a.state = determineAlarmStateV8_(prev,a);
    seen[a.stateId] = true;
  });
  Object.keys(previous).forEach(id => {
    const p = previous[id];
    if (!p.aktywny || seen[id]) return;
    alarms.push({
      aktywny:false,symbol:p.symbol,tf:p.tf,direction:p.direction,type:p.type||'ALARM WYGASŁ',
      signalCount:0,priority:p.priority||'OSTRZEŻENIE',severity:p.priority||'OSTRZEŻENIE',tfWeight:'',signals:[],
      rsi:'',mfi:'',macdPct:'',macdZ:'',fomo:'',reason:'Poprzedni aktywny alarm nie występuje już w bieżących danych.',
      ts:p.ts||'',freshness:'WYGASŁ',ageHours:'',limitHours:'',multiTf:p.tf==='WIELE TF',multiTfList:[],
      stateId:id,previousPriority:p.priority||'',state:'WYGASŁ',key:'WYGASŁ|'+id
    });
  });
  return alarms;
}

// V8 DUAL ENGINE — TACTICAL RISK CENTER V9
// Domknięcie tematu: centrum ryzyka Tactical.
// Łączy: exit risk, FOMO, strefę podaży/popytu, status Tactical,
// świeże alarmy ekstremów i zgodność wielu interwałów.
// Nie zmienia logiki LONG/Tactical i nie wykonuje transakcji.

function tacticalRiskAssessment_(asset, alarms) {
  const t = (asset || {}).timeframes || {};
  const z = (asset || {}).zone_context || {};
  const status = String((asset || {}).tactical_status || '');
  const exitRisk = Number((asset || {}).exit_risk_0_10);
  const fomoValues = ['1H','4H','1D'].map(tf => Number((t[tf] || {}).fomo_score_0_10)).filter(Number.isFinite);
  const maxFomo = fomoValues.length ? Math.max.apply(null, fomoValues) : 0;
  const active = (alarms || []).filter(a => a && a.aktywny && String(a.symbol) === String((asset || {}).symbol));
  const multiCritical = active.some(a => a.multiTf && a.priority === 'KRYTYCZNY');
  const critical = active.some(a => a.priority === 'KRYTYCZNY');
  const strong = active.some(a => a.priority === 'MOCNY');
  const overheatingMulti = active.some(a => a.multiTf && a.direction === 'PRZEGRZANIE');
  const washoutMulti = active.some(a => a.multiTf && a.direction === 'WYPRZEDANIE');
  const inSupply = z.in_supply === true;
  const inDemand = z.in_demand === true;

  let points = 0;
  const reasons = [];

  if (Number.isFinite(exitRisk)) {
    if (exitRisk >= 8) { points += 5; reasons.push('ryzyko wyjścia '+exitRisk+'/10'); }
    else if (exitRisk >= 6) { points += 3; reasons.push('ryzyko wyjścia '+exitRisk+'/10'); }
    else if (exitRisk >= 4) { points += 2; reasons.push('ryzyko wyjścia '+exitRisk+'/10'); }
  }

  if (maxFomo >= 9) { points += 4; reasons.push('FOMO '+maxFomo+'/10'); }
  else if (maxFomo >= 8) { points += 3; reasons.push('FOMO '+maxFomo+'/10'); }
  else if (maxFomo >= 7) { points += 1; reasons.push('FOMO '+maxFomo+'/10'); }

  if (inSupply) { points += 2; reasons.push('strefa podaży'); }
  if (inDemand) { points -= 1; reasons.push('strefa popytu'); }

  if (multiCritical) { points += 4; reasons.push('krytyczna zgodność wielu TF'); }
  else if (critical) { points += 3; reasons.push('krytyczny alarm'); }
  else if (strong) { points += 1; reasons.push('mocny alarm'); }

  if (overheatingMulti) { points += 2; reasons.push('przegrzanie wielu TF'); }
  if (washoutMulti) { points -= 1; reasons.push('wyprzedanie wielu TF'); }

  if (['NO_TRADE','NO_TRADE_FOMO','EXIT_RISK_HIGH'].indexOf(status) >= 0) {
    points += 3;
    reasons.push('status '+translateTactical_(status));
  } else if (['WATCH_SUPPLY','TAKE_PROFIT_WATCH'].indexOf(status) >= 0) {
    points += 1;
    reasons.push('status '+translateTactical_(status));
  }

  points = Math.max(0, Math.min(10, points));
  let state = 'NORMALNIE';
  if (points >= 8 || (multiCritical && overheatingMulti) || (Number.isFinite(exitRisk) && exitRisk >= 9)) state = 'OCHRONA KAPITAŁU';
  else if (points >= 5 || multiCritical || (Number.isFinite(exitRisk) && exitRisk >= 7)) state = 'PODWYŻSZONE RYZYKO';
  else if (points >= 2 || strong || critical) state = 'UWAGA';

  return {
    state: state,
    points: points,
    reason: reasons.length ? reasons.join(' | ') : 'brak istotnych czynników ryzyka',
    maxFomo: maxFomo,
    exitRisk: Number.isFinite(exitRisk) ? exitRisk : '',
    inSupply: inSupply,
    inDemand: inDemand
  };
}

function tacticalRiskIcon_(state) {
  if (state === 'OCHRONA KAPITAŁU') return '🛡️';
  if (state === 'PODWYŻSZONE RYZYKO') return '🔴';
  if (state === 'UWAGA') return '🟠';
  return '🟢';
}

function buildTacticalRiskMap_(tacticalJson, alarms) {
  const out = {};
  ((tacticalJson || {}).assets || []).forEach(a => {
    out[a.symbol] = tacticalRiskAssessment_(a, alarms);
  });
  return out;
}

function applyRiskCenterToTacticalSheet_(sheet, tacticalJson, alarms) {
  if (!sheet || sheet.getLastRow() < 1) return;
  const riskMap = buildTacticalRiskMap_(tacticalJson, alarms);
  const values = sheet.getDataRange().getValues();
  const head = values[0].map(String);
  const symIdx = head.indexOf('AKTYWO');
  if (symIdx < 0) return;

  let stateIdx = head.indexOf('CENTRUM RYZYKA');
  let reasonIdx = head.indexOf('POWÓD RYZYKA');
  if (stateIdx < 0) { stateIdx = head.length; head.push('CENTRUM RYZYKA'); }
  if (reasonIdx < 0) { reasonIdx = head.length; head.push('POWÓD RYZYKA'); }

  const width = head.length;
  const out = [head];
  for (let i=1; i<values.length; i++) {
    const row = values[i].slice();
    while (row.length < width) row.push('');
    const sym = String(row[symIdx] || '').trim();
    const r = riskMap[sym];
    if (r) {
      row[stateIdx] = tacticalRiskIcon_(r.state)+' '+r.state+' ('+r.points+'/10)';
      row[reasonIdx] = r.reason;
    }
    out.push(row);
  }

  sheet.clearContents();
  sheet.getRange(1,1,out.length,width).setValues(out);
  sheet.setFrozenRows(1);
  sheet.getRange(1,1,1,width).setFontWeight('bold');
  if (out.length > 1) sheet.getRange(2,1,out.length-1,width).setWrap(true).setVerticalAlignment('middle');
  sheet.setColumnWidth(stateIdx+1,190);
  sheet.setColumnWidth(reasonIdx+1,390);

  const riskRange = sheet.getRange(2,stateIdx+1,Math.max(1,out.length-1),1);
  const rules = (sheet.getConditionalFormatRules() || []).filter(r => !r.getRanges().some(x => x.getColumn() === stateIdx+1));
  rules.push(SpreadsheetApp.newConditionalFormatRule().whenTextContains('OCHRONA KAPITAŁU').setBackground('#f4cccc').setFontColor('#990000').setBold(true).setRanges([riskRange]).build());
  rules.push(SpreadsheetApp.newConditionalFormatRule().whenTextContains('PODWYŻSZONE RYZYKO').setBackground('#fce5cd').setFontColor('#b45f06').setBold(true).setRanges([riskRange]).build());
  rules.push(SpreadsheetApp.newConditionalFormatRule().whenTextContains('UWAGA').setBackground('#fff2cc').setFontColor('#7f6000').setRanges([riskRange]).build());
  rules.push(SpreadsheetApp.newConditionalFormatRule().whenTextContains('NORMALNIE').setBackground('#d9ead3').setFontColor('#274e13').setRanges([riskRange]).build());
  sheet.setConditionalFormatRules(rules);
}

function updateMainPanelExtremeSummary_(sheet, alarms) {
  if (!sheet || sheet.getLastRow() < 2) return;
  const ss = sheet.getParent();
  let tacticalJson = {assets:[]};
  try { tacticalJson = JSON.parse(fetchText_(V8_FILES.tactical)); } catch (e) {}
  const riskMap = buildTacticalRiskMap_(tacticalJson, alarms);

  applyRiskCenterToTacticalSheet_(ss.getSheetByName(V8_SHEETS.tactical), tacticalJson, alarms);

  const bySymbol = {};
  (alarms || []).forEach(a => { if (panelEligibleExtreme_(a)) (bySymbol[a.symbol] = bySymbol[a.symbol] || []).push(a); });
  const values = sheet.getDataRange().getValues();
  const head = values[0].map(String), symIdx = head.indexOf('AKTYWO'), alertIdx = head.indexOf('ALERT');
  if (symIdx < 0 || alertIdx < 0) return;

  for (let i=1; i<values.length; i++) {
    const sym = String(values[i][symIdx] || '').trim();
    if (!sym) continue;
    const best = selectMainPanelExtreme_(bySymbol[sym] || []);
    const risk = riskMap[sym];
    let base = String(values[i][alertIdx] || '').trim();
    base = base.replace(/\s*\|\s*[🟡🟠🔴🛡️🟢]\s*(OSTRZEŻENIE|MOCNY|KRYTYCZNY|NORMALNIE|UWAGA|PODWYŻSZONE RYZYKO|OCHRONA KAPITAŁU)[^|]*/g,'').trim();

    const extras = [];
    if (best) {
      const icon = best.priority === 'KRYTYCZNY' ? '🔴' : best.priority === 'MOCNY' ? '🟠' : '🟡';
      const stateSuffix = best.state && best.state !== 'BEZ ZMIAN' ? ' '+best.state : '';
      extras.push(best.multiTf ? icon+' '+best.priority+' WIELE TF — '+best.direction+' '+best.multiTfList.join('+')+stateSuffix : icon+' '+best.priority+' '+best.tf+' — '+best.signalCount+' sygnał(y)'+stateSuffix);
    }
    if (risk && risk.state !== 'NORMALNIE') extras.push(tacticalRiskIcon_(risk.state)+' '+risk.state+' '+risk.points+'/10');

    const cleanBase = (!base || base === 'BRAK') ? '' : base;
    values[i][alertIdx] = [cleanBase].concat(extras).filter(Boolean).join(' | ') || 'BRAK';
  }

  sheet.getRange(1,1,values.length,values[0].length).setValues(values);
  formatPanel_(sheet);
  formatMainPanelAlertsV9_(sheet);

  (tacticalJson.assets || []).forEach(a => tacticalMap[String(a.symbol)] = a);
  const entries = Object.keys(meta).filter(s => s !== 'BTC').map(sym => {
    const m = meta[sym], t=tacticalMap[sym] || {};
    return {
      sym:sym,
      status:translateTactical_(m.prod_status || t.tactical_status || ''),
      meta:translateMetaV10_(m.meta_decision || ''),
      risk:numV10_(m.risk_score, t.exit_risk_0_10),

      cand:yesNoV10_(m.challenger_candidate),
      reason:translateReasonV10_(m.reasons || m.reason || '')
    };
  });
  const regime = entries.length && meta[entries[0].sym] ? translateRegimeV10_(meta[entries[0].sym].regime_gate || '') : '—';

    else counts.WSTRZYMAJ++;
  });

  sheet.getRange('D4:I6').setValues([
    ['ZEZWÓL',counts.ZEZWOL,'OSTROŻNIE',counts.OSTROZNIE,'BLOKUJ',counts.BLOKUJ],
    ['WSTRZYMAJ',counts.WSTRZYMAJ,'ŚR. RYZYKO',Math.round(avgRisk*10)/10,'FOMO ≥ 8',entries.filter(e=>e.fomo>=8).length],
    ['PRZEGRZANIE ≥ 3',entries.filter(e=>e.heat>=3).length,'BEZ TRANSAKCJI',entries.filter(e=>e.status==='NIE GRAJ').length,'OBSERWUJ PODAŻ',entries.filter(e=>e.status==='OBSERWUJ PODAŻ').length]

  styleMorningBriefV10_(sheet, detail.length);

function parseMetaCsvV10_(csvText) {
  const rows = Utilities.parseCsv(csvText || '');
  if (!rows.length) return {};
  const head = rows[0].map(x=>String(x).trim());
  const out = {};
  rows.slice(1).forEach(r => {
    const obj = {}; head.forEach((h,i)=>obj[h]=r[i]);
    const sym = String(obj.symbol || obj.AKTYWO || '').trim();
    if (sym) out[sym] = obj;
  });
  return out;
}

function translateMetaV10_(v) {
  const m = {'ALLOW':'ZEZWÓL','CAUTION':'OSTROŻNIE','BLOCK':'BLOKUJ','HOLD_CHALLENGER':'WSTRZYMAJ KANDYDATA'};
  return m[String(v||'').toUpperCase()] || String(v||'').replaceAll('_',' ');
}
function translateRegimeV10_(v) {
  const m={'ACTIVE':'AKTYWNY','CAUTION':'OSTROŻNY','BLOCK':'BLOKADA'};
  return m[String(v||'').toUpperCase()] || String(v||'').replaceAll('_',' ');
}
function translateReasonV10_(v) {
  return String(v||'')
    .replaceAll('exit risk>=6','ryzyko wyjścia ≥ 6')
    .replaceAll('exit risk>=4','ryzyko wyjścia ≥ 4')
    .replaceAll('status SUPPLY','status: podaż')
    .replaceAll('status NO_TRADE','status: bez transakcji')
    .replaceAll('FOMO>=8','FOMO ≥ 8')
    .replaceAll('FOMO>=6','FOMO ≥ 6');
}
function maxFomoV10_(a) {
  const t=(a||{}).timeframes||{};
  return Math.max(0, Number((t['1H']||{}).fomo_score_0_10||0), Number((t['4H']||{}).fomo_score_0_10||0), Number((t['1D']||{}).fomo_score_0_10||0));
}
function numV10_(a,b) { const n=Number(a); if (Number.isFinite(n)) return n; const m=Number(b); return Number.isFinite(m)?m:0; }
function yesNoV10_(v) { const s=String(v||'').toLowerCase(); return ['true','1','yes','tak'].includes(s) ? 'TAK' : 'NIE'; }

function morningRadarModeV10_(sheet) {

function morningRadarTfV10_(asset, mode) {
  if (!asset) return {};
  if (mode === 'AUTO') return asset.auto || {};
  return ((asset.timeframes || {})[mode]) || {};
}

function round1V10_(v) {
  const n = Number(v);
  return Number.isFinite(n) ? Math.round(n * 10) / 10 : '';
}

function macdBriefV10_(v) {
  const s = String(v || '').toUpperCase();
  if (s === 'EXTREME_POSITIVE') return 'EKSTREMUM +';
  if (s === 'EXTREME_NEGATIVE') return 'EKSTREMUM −';
  if (s === 'WARNING_POSITIVE') return 'UWAGA +';
  if (s === 'WARNING_NEGATIVE') return 'UWAGA −';
  if (s === 'NO_DATA') return 'BRAK';
  return 'NORMALNY';
}

function onSelectionChange(e) {
  try {
    if (!e || !e.range) return;
    const sheet = e.range.getSheet();
    if (sheet.getName() !== V10_MORNING_SHEET) return;
    if (e.range.getRow() !== 2) return;
    const col = e.range.getColumn();
    const modeMap = {2:'1H',3:'2H',4:'4H',5:'AUTO'};
    const mode = modeMap[col];
    if (!mode) return;
    if (morningRadarModeV10_(sheet) === mode) {
      styleMorningRadarButtonsV10_(sheet, mode);
      return;
    }

    sheet.getRange('K1').setValue(mode);
    styleMorningRadarButtonsV10_(sheet, mode);

    const cached = PropertiesService.getDocumentProperties().getProperty('V10_MORNING_RADAR_CACHE');
    if (cached) {
      writeMorningRadarBrief_(sheet, JSON.parse(cached));
      SpreadsheetApp.getActiveSpreadsheet().toast('Radar przełączony na ' + mode, 'V8', 2);
    } else {
      SpreadsheetApp.getActiveSpreadsheet().toast('Wybrano ' + mode + '. Uruchom ODŚWIEŻ PORANNY BRIEF jeden raz.', 'V8', 4);
    }
  } catch (err) {
    console.log('onSelectionChange radar: ' + err);
  }
}

function radarKindV10_(status) {
  const s = String(status || '').toUpperCase();
  if (s.indexOf('GORĄCO') >= 0) return 'HOT';
  if (s.indexOf('CHŁODNO') >= 0) return 'COLD';
  if (s.indexOf('UWAGA') >= 0) return 'WARN';
  if (s.indexOf('NEUTRAL') >= 0) return 'NEUTRAL';
  return 'NORMAL';
}

function strongestMorningRowKindV10_(sheet, r) {
  const status = String(sheet.getRange(r,8).getDisplayValue() || '').toUpperCase();
  const macd = String(sheet.getRange(r,5).getDisplayValue() || '').toUpperCase();
  const rsi = Number(sheet.getRange(r,3).getValue());
  const mfi = Number(sheet.getRange(r,4).getValue());
  const fomo = Number(sheet.getRange(r,6).getValue());

  // Najpierw pełny stan radaru.
  if (status.indexOf('GORĄCO') >= 0) return 'HOT';
  if (status.indexOf('CHŁODNO') >= 0) return 'COLD';
  if (status.indexOf('UWAGA') >= 0) return 'WARN';

  // Jeśli radar jest NEUTRALNY, ale którykolwiek wskaźnik już daje sygnał,
  // zielone tło nie może sugerować "wszystko w normie".
  if (Number.isFinite(fomo) && fomo >= 8) return 'FOMO';
  if (macd.indexOf('EKSTREMUM') >= 0 || macd.indexOf('UWAGA') >= 0) return 'WARN';
  if (Number.isFinite(rsi) && (rsi >= 75 || rsi <= 25)) return 'WARN';
  if (Number.isFinite(mfi) && (mfi >= 90 || mfi <= 10)) return 'WARN';

  return radarKindV10_(status);
}

// ===== V10 SHARED INDICATOR LANGUAGE =====
// Jedna warstwa języka dla wszystkich widoków V8.
// Nie zmienia progów ani obliczeń Tactical/LONG — ujednolica tylko znaczenie i opis sygnałów.
function macdDisplayLabelV10_(x) {
  const src = String((x && x.macd_extreme_label) || 'NORMAL').toUpperCase();
  const hist = Number(x && x.macd_hist);
  if (src === 'NORMAL' || src === 'NO_DATA' || !Number.isFinite(hist)) return 'NORMALNY';
  const extreme = src.indexOf('EXTREME') >= 0;
  const warning = src.indexOf('WARNING') >= 0;
  if (!extreme && !warning) return 'NORMALNY';
  if (hist > 0) return extreme ? 'MACD EKSTREMUM + — GÓRNE GRANICE' : 'MACD UWAGA + — GÓRNA STREFA';
  if (hist < 0) return extreme ? 'MACD EKSTREMUM − — DOLNE GRANICE' : 'MACD UWAGA − — DOLNA STREFA';
  return 'NORMALNY';
}

function extremeLabel_(x, cfg) {
  const rsi = Number(x.rsi14), mfi = Number(x.mfi14), fomo = Number(x.fomo_score_0_10), labels = [];
  if (Number.isFinite(rsi) && rsi >= cfg.rsiHigh) labels.push(rsi >= 80 ? 'RSI GÓRNE GRANICE' : 'RSI GÓRNA STREFA');
  if (Number.isFinite(rsi) && rsi <= cfg.rsiLow) labels.push(rsi <= 20 ? 'RSI DOLNE GRANICE' : 'RSI DOLNA STREFA');
  if (Number.isFinite(mfi)) {
    if (mfi >= cfg.mfiExtremeHigh) labels.push('MFI GÓRNE GRANICE');
    else if (mfi >= cfg.mfiWarnHigh) labels.push('MFI GÓRNA STREFA');
    if (mfi <= cfg.mfiExtremeLow) labels.push('MFI DOLNE GRANICE');
    else if (mfi <= cfg.mfiWarnLow) labels.push('MFI DOLNA STREFA');
  }
  const macd = macdDisplayLabelV10_(x);
  if (macd !== 'NORMALNY') labels.push(macd);
  if (Number.isFinite(fomo) && fomo >= cfg.fomoHard) labels.push(fomo >= 9 ? 'FOMO GÓRNE GRANICE' : 'FOMO GÓRNA STREFA');
  return labels.length ? labels.join(' + ') : 'BRAK';
}

function alarmDirection_(a) {
  const signals = a.signals || [];
  let hot = 0, cold = 0;
  signals.forEach(s => {
    const x = String(s || '').toUpperCase();
    if (x.indexOf('GÓRNE') >= 0 || x.indexOf('GÓRNA') >= 0 || x.indexOf('FOMO') >= 0) hot++;
    if (x.indexOf('DOLNE') >= 0 || x.indexOf('DOLNA') >= 0) cold++;
    // zgodność wsteczna ze starszym nazewnictwem podczas jednego cyklu migracji
    if (x.indexOf('WYKUPIENIE') >= 0 || x.indexOf('DODATNIE') >= 0) hot++;
    if (x.indexOf('WYPRZEDANIE') >= 0 || x.indexOf('UJEMNE') >= 0) cold++;
  });
  if (hot > 0 && cold === 0) return 'PRZEGRZANIE';
  if (cold > 0 && hot === 0) return 'WYPRZEDANIE';
  return 'MIESZANE';
}

function buildExtremeAlarms_(ss, tacticalJson) {
  const cfg = getExtremeSettings_(ss);
  if (!cfg.enabled) return [];
  const base = [];
  (tacticalJson.assets || []).forEach(a => {
    const t = a.timeframes || {};
    ['1H','4H','1D'].forEach(tf => {
      const x = t[tf] || {};
      const rsi = Number(x.rsi14), mfi = Number(x.mfi14), fomo = Number(x.fomo_score_0_10);
      const macdPct = Number(x.macd_hist_percentile), macdZ = Number(x.macd_hist_zscore);
      const ts = x.last_close_time_utc || '';
      const signals = [];

      if (Number.isFinite(rsi) && rsi >= cfg.rsiHigh) signals.push(rsi >= 80 ? 'RSI GÓRNE GRANICE' : 'RSI GÓRNA STREFA');
      if (Number.isFinite(rsi) && rsi <= cfg.rsiLow) signals.push(rsi <= 20 ? 'RSI DOLNE GRANICE' : 'RSI DOLNA STREFA');

      if (Number.isFinite(mfi)) {
        if (mfi >= cfg.mfiExtremeHigh) signals.push('MFI GÓRNE GRANICE');
        else if (mfi >= cfg.mfiWarnHigh) signals.push('MFI GÓRNA STREFA');
        if (mfi <= cfg.mfiExtremeLow) signals.push('MFI DOLNE GRANICE');
        else if (mfi <= cfg.mfiWarnLow) signals.push('MFI DOLNA STREFA');
      }

      const macd = macdDisplayLabelV10_(x);
      if (macd !== 'NORMALNY') signals.push(macd);

      if (Number.isFinite(fomo) && fomo >= cfg.fomoHard) signals.push(fomo >= 9 ? 'FOMO GÓRNE GRANICE' : 'FOMO GÓRNA STREFA');
      if (!signals.length) return;

      const count = signals.length;
      const priority = confluencePriorityForTf_(count, tf);
      const tfWeight = timeframeWeight_(tf);
      const freshness = alarmFreshness_(tf, ts, cfg);
      const n = count === 1 ? '1 sygnał' : count < 5 ? count+' sygnały' : count+' sygnałów';
      const reason = a.symbol+' '+tf+': '+n+' zgodne. Priorytet '+priority+', waga '+tfWeight+', świeżość '+freshness.status+(freshness.ageHours === '' ? '' : ' ('+freshness.ageHours+'h / limit '+freshness.limitHours+'h)')+'. '+signals.join(' + ')+'.';
      base.push({
        aktywny:freshness.fresh,symbol:a.symbol,tf:tf,type:signals.join(' + '),signalCount:count,
        priority:priority,severity:priority,tfWeight:tfWeight,signals:signals,
        rsi:Number.isFinite(rsi)?rsi:'',mfi:Number.isFinite(mfi)?mfi:'',macdPct:Number.isFinite(macdPct)?macdPct:'',
        macdZ:Number.isFinite(macdZ)?macdZ:'',fomo:Number.isFinite(fomo)?fomo:'',reason:reason,ts:ts,
        freshness:freshness.status,ageHours:freshness.ageHours,limitHours:freshness.limitHours,
        key:[a.symbol,tf,signals.join(' + ')].join('|')
      });
    });
  });
  return base.concat(buildMultiTfConfluence_(base)).sort(compareExtremeFreshnessPriorityV7_);
}

    if (!sh) return;
    sh.setHiddenGridlines(true);
    const rows = Math.max(1, sh.getLastRow());
    const cols = Math.max(1, sh.getLastColumn());
    const range = sh.getRange(1,1,rows,cols);
    range.setFontFamily('Roboto').setFontColor(V8_UI.text).setVerticalAlignment('middle');
    if (name !== 'PORANNY_BRIEF') {
      range.setBackground(V8_UI.canvas);
      const h = sh.getRange(1,1,1,cols);
      h.setBackground(V8_UI.header).setFontWeight('bold').setHorizontalAlignment('center');
      styleUnifiedV8Cells_(sh, rows, cols);
    }
  });
}

function styleUnifiedV8Cells_(sh, rows, cols) {
  if (rows < 1 || cols < 1) return;
  const vals = sh.getRange(1,1,rows,cols).getDisplayValues();
  const head = vals[0].map(x => String(x||'').toUpperCase());
  const bg = Array.from({length:rows}, () => Array(cols).fill(V8_UI.canvas));
  const fg = Array.from({length:rows}, () => Array(cols).fill(V8_UI.text));
  const fw = Array.from({length:rows}, () => Array(cols).fill('normal'));
  for (let c=0;c<cols;c++) { bg[0][c]=V8_UI.header; fw[0][c]='bold'; }

  for (let r=1;r<rows;r++) {
    for (let c=0;c<cols;c++) {
      const raw = String(vals[r][c]||'').trim();
      const t = raw.toUpperCase();
      const h = head[c] || '';
      let style = unifiedSemanticStyleV10_(t, raw, h);
      if (style) {
        bg[r][c] = style.bg; fg[r][c] = style.fg; fw[r][c] = 'bold';
      }
    }
  }
  sh.getRange(1,1,rows,cols).setBackgrounds(bg).setFontColors(fg).setFontWeights(fw);
}

function unifiedSemanticStyleV10_(t, raw, header) {
  const num = Number(String(raw).replace(',','.'));

  // Najpierw konkretna semantyka górnych/dolnych zakresów.
  if (t.includes('GÓRNE GRANICE') || t.includes('EKSTREMUM +') || t.includes('SKRAJNIE +'))
    return {bg:V8_UI.hot,fg:V8_UI.hotText};
  if (t.includes('DOLNE GRANICE') || t.includes('EKSTREMUM −') || t.includes('EKSTREMUM -') || t.includes('SKRAJNIE -'))
    return {bg:V8_UI.cold,fg:V8_UI.coldText};
  if (t.includes('GÓRNA STREFA') || t.includes('UWAGA +'))
    return {bg:V8_UI.warn,fg:V8_UI.warnText};
  if (t.includes('DOLNA STREFA') || t.includes('UWAGA -'))
    return {bg:V8_UI.cold,fg:V8_UI.coldText};

  // Wspólny język statusów.
  if (t === 'GORĄCO' || t.includes('BLOKUJ') || t.includes('KRYTYCZNY') || t.includes('NIE DOKŁADAJ'))
    return {bg:V8_UI.hot,fg:V8_UI.hotText};
  if (t === 'CHŁODNO' || t.includes('WYPRZEDANIE'))
    return {bg:V8_UI.cold,fg:V8_UI.coldText};
  if (t.includes('NEUTRALNIE') || t === 'KUP' || t.includes('KUP TRENDOWO') || t === 'ZEZWÓL')
    return {bg:V8_UI.neutral,fg:V8_UI.neutralText};
  if (t.includes('OSTROŻNIE') || t.includes('OSTRZEŻENIE') || t === 'CZEKAJ' || t.includes('SŁABNIE'))
    return {bg:V8_UI.warn,fg:V8_UI.warnText};
  if (t.includes('FOMO') && !t.includes('FOMO 0'))
    return {bg:V8_UI.fomo,fg:V8_UI.fomoText};
  if (t.includes('MOCNY') || t.includes('SPRAWDŹ') || t.includes('NASILA SIĘ'))
    return {bg:V8_UI.strong,fg:V8_UI.strongText};
  if (t.includes('MACD ALERT') || t.includes('INFO') || t.includes('BEZ ZMIAN'))
    return {bg:V8_UI.info,fg:V8_UI.infoText};

  // Liczbowe progi RSI/MFI/FOMO w silnikach — dokładnie te same progi, bez zmiany logiki.
  if (Number.isFinite(num)) {
    if (header.includes('RSI')) {
      if (num >= 80) return {bg:V8_UI.hot,fg:V8_UI.hotText};
      if (num >= 75) return {bg:V8_UI.warn,fg:V8_UI.warnText};
      if (num <= 20) return {bg:V8_UI.cold,fg:V8_UI.coldText};
      if (num <= 25) return {bg:V8_UI.cold,fg:V8_UI.coldText};
    }
    if (header.includes('MFI')) {
      if (num >= 90) return {bg:V8_UI.hot,fg:V8_UI.hotText};
      if (num >= 80) return {bg:V8_UI.warn,fg:V8_UI.warnText};
      if (num <= 10) return {bg:V8_UI.cold,fg:V8_UI.coldText};
      if (num <= 20) return {bg:V8_UI.cold,fg:V8_UI.coldText};
    }
    if (header.includes('FOMO')) {
      if (num >= 8) return {bg:V8_UI.fomo,fg:V8_UI.fomoText};
    }
    if (header.includes('RYZYKO')) {
      if (num >= 8) return {bg:V8_UI.hot,fg:V8_UI.hotText};
      if (num >= 5) return {bg:V8_UI.warn,fg:V8_UI.warnText};
      if (num <= 2) return {bg:V8_UI.neutral,fg:V8_UI.neutralText};
    }
  }
  return null;
}

// Tylko warstwa wizualna. Nie zmienia logiki LONG/TACTICAL, progów ani danych w maszynowni.

const V10_UI = {
  hot: '#FAD1D1', hotText: '#8F2D2D',
  cold: '#D9EAF7', coldText: '#245B78',
  neutral: '#DDF2E3', neutralText: '#1F6B3A',
  warn: '#FFF0B8', warnText: '#7A5B00',
  fomo: '#FCE2B8', fomoText: '#8A4B08',
  macd: '#E8EDF2', macdText: '#44546A',
  info: '#EAF5FB', infoText: '#2F5F77'
};

function applyUnifiedThemeV10_(ss) {
  const names = [
    'PANEL','PORTFEL_LONG','PORTFEL_TACTICAL','SILNIK_LONG','SILNIK_TACTICAL',
    'PRZEPLYWY','ETF_BTC_ETH','ETF_HISTORIA','LAB_ANALIZA_AKTYWA','MARKET_READER',
    'ALARMY_EKSTREMOW','HISTORIA','PORANNY_BRIEF'
  ];
  names.forEach(name => {
    const sh = ss.getSheetByName(name);
    if (!sh) return;
    applyUnifiedSheetThemeV10_(sh);
  });
}

function applyUnifiedSheetThemeV10_(sheet) {
  const range = sheet.getDataRange();
  if (!range || range.getNumRows() < 1 || range.getNumColumns() < 1) return;
  const vals = range.getDisplayValues();
  const merged = range.getMergedRanges();
  const mergeMap = {};
  merged.forEach(m => {
    const r0 = m.getRow(), c0 = m.getColumn(), rn = m.getNumRows(), cn = m.getNumColumns();
    for (let rr = r0; rr < r0 + rn; rr++) {
      for (let cc = c0; cc < c0 + cn; cc++) mergeMap[rr + ':' + cc] = m;
    }
  });
  const styledMerged = {};

  for (let r = 0; r < vals.length; r++) {
    for (let c = 0; c < vals[r].length; c++) {
      const raw = String(vals[r][c] || '').trim();
      if (!raw) continue;
      const t = raw.toUpperCase();
      const rr = r + 1, cc = c + 1;
      const mergedRange = mergeMap[rr + ':' + cc];
      let cell = sheet.getRange(rr, cc);
      if (mergedRange) {
        const key = mergedRange.getA1Notation();
        if (styledMerged[key]) continue;
        styledMerged[key] = true;
        cell = mergedRange;
      }

      if (isHotV10_(t)) styleBriefCardV10_(cell, V10_UI.hot, V10_UI.hotText);
      else if (isColdV10_(t)) styleBriefCardV10_(cell, V10_UI.cold, V10_UI.coldText);
      else if (isNeutralV10_(t)) styleBriefCardV10_(cell, V10_UI.neutral, V10_UI.neutralText);
      else if (isWarnV10_(t)) styleBriefCardV10_(cell, V10_UI.warn, V10_UI.warnText);
      else if (isFomoV10_(t)) styleBriefCardV10_(cell, V10_UI.fomo, V10_UI.fomoText);
      else if (isMacdV10_(t)) styleBriefCardV10_(cell, V10_UI.macd, V10_UI.macdText);
      else if (isInfoV10_(t)) styleBriefCardV10_(cell, V10_UI.info, V10_UI.infoText);

      const num = Number(String(raw).replace(',', '.'));
      if (Number.isFinite(num)) {
        const metric = nearestMetricHeaderV10_(vals, r, c);
        if (metric === 'RSI') {
          if (num >= 75) styleBriefCardV10_(cell, V10_UI.hot, V10_UI.hotText);
          else if (num <= 25) styleBriefCardV10_(cell, V10_UI.cold, V10_UI.coldText);
        } else if (metric === 'MFI') {
          if (num >= 80) styleBriefCardV10_(cell, V10_UI.hot, V10_UI.hotText);
          else if (num <= 20) styleBriefCardV10_(cell, V10_UI.cold, V10_UI.coldText);
        } else if (metric.indexOf('FOMO') >= 0) {
          if (num >= 8) styleBriefCardV10_(cell, V10_UI.fomo, V10_UI.fomoText);
          else if (num >= 6) styleBriefCardV10_(cell, V10_UI.warn, V10_UI.warnText);
        } else if (metric.indexOf('RYZYKO') >= 0) {
          if (num >= 8) styleBriefCardV10_(cell, V10_UI.hot, V10_UI.hotText);
          else if (num >= 5) styleBriefCardV10_(cell, V10_UI.warn, V10_UI.warnText);
          else if (num <= 2) styleBriefCardV10_(cell, V10_UI.neutral, V10_UI.neutralText);
        } else if (metric.indexOf('PRZEGRZ') >= 0) {
          if (num >= 7) styleBriefCardV10_(cell, V10_UI.hot, V10_UI.hotText);
          else if (num >= 3) styleBriefCardV10_(cell, V10_UI.fomo, V10_UI.fomoText);
        }
      }
    }
  }
}

function nearestMetricHeaderV10_(vals, r, c) {
  const recognized = ['RSI','MFI','FOMO','RYZYKO','RYZYKO WYJŚCIA','PRZEGRZANIE','PRZEGRZ.'];
  for (let rr = r - 1; rr >= Math.max(0, r - 6); rr--) {
    const h = String((vals[rr] || [])[c] || '').trim().toUpperCase();
    if (recognized.some(x => h === x || h.indexOf(x) >= 0)) return h;
  }
  return '';
}

function isHotV10_(t) {
  return t === 'GORĄCO' || t === 'BLOKUJ' || t === 'KRYTYCZNY' ||
         t === 'OCHRONA KAPITAŁU' || t === 'NIE DOKŁADAJ' ||
         t.indexOf('EKSTREMUM +') >= 0 || t.indexOf('GÓRNE GRANICE') >= 0;
}
function isColdV10_(t) {
  return t === 'CHŁODNO' || t === 'WYPRZEDANIE' ||
         t.indexOf('EKSTREMUM −') >= 0 || t.indexOf('EKSTREMUM -') >= 0 ||
         t.indexOf('DOLNE GRANICE') >= 0 || t.indexOf('DOLNE OSTRZEŻ') >= 0;
}
function isNeutralV10_(t) {
  return t === 'NEUTRALNIE' || t === 'ZEZWÓL' || t === 'GOTOWE' ||
         t === 'AKTUALNE' || t === 'AKTUALNY' || t === 'KUP' || t === 'KUP TRENDOWO';
}
function isWarnV10_(t) {
  return t === 'UWAGA' || t.indexOf('UWAGA ') === 0 || t === 'OSTROŻNIE' ||
         t === 'OSTROŻNY' || t === 'MOCNY' || t === 'CZEKAJ' ||
         t === 'PODWYŻSZONE RYZYKO' || t === 'NASILA SIĘ';
}
function isFomoV10_(t) {
  return t.indexOf('FOMO') >= 0 && (t.indexOf('8') >= 0 || t.indexOf('HARD') >= 0 || t.indexOf('BLOK') >= 0);
}
function isMacdV10_(t) {
  return t.indexOf('MACD') >= 0 || t === 'BEZ ZMIAN' || t.indexOf('WSTRZYMAJ') >= 0;
}
function isInfoV10_(t) {
  return t === 'INFO' || t.indexOf('STREF') >= 0 || t.indexOf('DCA') >= 0;
}
