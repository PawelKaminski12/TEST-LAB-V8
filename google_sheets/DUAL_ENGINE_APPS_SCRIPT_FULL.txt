function onOpen() {

    .addItem('ODŚWIEŻ_WSZYSTKO', 'ODSWIEZ_WSZYSTKO')

function V8_SETUP() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  Object.values(V8_SHEETS).forEach(name => getOrCreateSheet_(ss, name));
  setupPortfolioLong_(ss.getSheetByName(V8_SHEETS.portfolioLong));
  setupPortfolioTactical_(ss.getSheetByName(V8_SHEETS.portfolioTactical));
  setupSettings_(ss.getSheetByName(V8_SHEETS.settings));
  setupExtremes_(ss.getSheetByName(V8_SHEETS.extremes));
  formatPanel_(ss.getSheetByName(V8_SHEETS.panel));
  ss.toast('Układ V8 DUAL ENGINE gotowy', 'V8', 4);
}

function ODSWIEZ_WSZYSTKO() {
  const lock = LockService.getDocumentLock();
  if (!lock.tryLock(1000)) throw new Error('Odświeżanie już trwa.');
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    V8_SETUP();

    const oldPanel = readPanelState_(ss.getSheetByName(V8_SHEETS.panel));
    const oldAlarms = readActiveExtremeKeys_(ss.getSheetByName(V8_SHEETS.extremes));

    const panelCsv = fetchText_(V8_FILES.panel);
    const machineCsv = fetchText_(V8_FILES.machine);
    const longJson = JSON.parse(fetchText_(V8_FILES.long));
    const tacticalJson = JSON.parse(fetchText_(V8_FILES.tactical));
    const readinessJson = JSON.parse(fetchText_(V8_FILES.tacticalReadiness));
    const phaseJson = JSON.parse(fetchText_(V8_FILES.marketPhase));
    const snapshotJson = JSON.parse(fetchText_(V8_FILES.snapshot));
    const labJson = JSON.parse(fetchText_(V8_FILES.labReport));
    const mrCsv = fetchText_(V8_FILES.marketReader);

    writePolishPanel_(ss.getSheetByName(V8_SHEETS.panel), panelCsv);
    writeLongEngine_(ss.getSheetByName(V8_SHEETS.long), longJson);
    writeTacticalEngine_(ss.getSheetByName(V8_SHEETS.tactical), tacticalJson, readinessJson, phaseJson);
    writeFlowEngine_(ss.getSheetByName(V8_SHEETS.flow), phaseJson, snapshotJson);
    writeEtfPanel_(ss.getSheetByName(V8_SHEETS.etf), phaseJson, snapshotJson);
    writeLabPanel_(ss.getSheetByName(V8_SHEETS.lab), labJson);
    writeCsvSheet_(ss.getSheetByName(V8_SHEETS.marketReader), mrCsv, false);

    const machineSheet = getOrCreateSheet_(ss, 'DANE_MASZYNOWNIA');
    writeCsvSheet_(machineSheet, machineCsv, false);
    machineSheet.hideSheet();

    const alarms = buildExtremeAlarms_(ss, tacticalJson);
    writeExtremeAlarms_(ss.getSheetByName(V8_SHEETS.extremes), alarms);
    appendExtremeHistory_(ss, oldAlarms, alarms);
    notifyNewExtremeAlarms_(ss, oldAlarms, alarms);

    formatPanel_(ss.getSheetByName(V8_SHEETS.panel));
    appendHistoryOnChanges_(ss, oldPanel, readPanelState_(ss.getSheetByName(V8_SHEETS.panel)));
    stampRefresh_(ss);

    const active = alarms.filter(a => a.aktywny).length;
    ss.toast('LONG + TACTICAL + ETF + LAB odświeżone. Aktywne ekstrema: ' + active, 'V8 DUAL ENGINE', 5);
  } finally {
    lock.releaseLock();
  }
}

function SPRAWDZ_EKSTREMA() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

function fetchText_(url) {
  const response = UrlFetchApp.fetch(url, {
    muteHttpExceptions: true,
    followRedirects: true,
    headers: {'Cache-Control': 'no-cache'}
  });
  const code = response.getResponseCode();
  if (code < 200 || code >= 300) throw new Error('HTTP ' + code + ' dla ' + url);
  return response.getContentText('UTF-8');
}

function getOrCreateSheet_(ss, name) {
  return ss.getSheetByName(name) || ss.insertSheet(name);
}

function writeCsvSheet_(sheet, csvText, preserveCompactPanel) {
  const rows = Utilities.parseCsv(csvText);
  sheet.clearContents();
  if (!rows.length) return;
  sheet.getRange(1, 1, rows.length, rows[0].length).setValues(rows);
  sheet.setFrozenRows(1);
  if (preserveCompactPanel) {
    const maxCols = 7;
    if (sheet.getMaxColumns() > maxCols) sheet.hideColumns(maxCols + 1, sheet.getMaxColumns() - maxCols);
  }
}

function writePolishPanel_(sheet, csvText) {
  const rows = Utilities.parseCsv(csvText);
  sheet.clearContents();
  if (!rows.length) return;
  const head = rows[0].map(String);
  const idx = {}; head.forEach((h,i) => idx[h] = i);
  const out = [['AKTYWO','CENA','LONG','TACTICAL','RYZYKO 0–10','ZGODNOŚĆ','ALERT']];
  rows.slice(1).forEach(r => {
    if (!String(r[idx['AKTYWO']] || '').trim()) return;
    out.push([
      r[idx['AKTYWO']], r[idx['CENA']], translateLong_(r[idx['LONG']]),
      translateTactical_(r[idx['TACTICAL']]), r[idx['RYZYKO_0_10']],
      translateAgreement_(r[idx['ZGODNOSC']]), translateAlert_(r[idx['ALERT']])
    ]);
  });
  sheet.getRange(1,1,out.length,out[0].length).setValues(out);
  sheet.setFrozenRows(1);
}

function writeLongEngine_(sheet, data) {
  const headers = ['AKTYWO','JAKOŚĆ','STATUS DANYCH','STREFY','TREND','FOMO','ROTACJA','FAZA RYNKU','FLOW LONG 0–10','FLOW OCENA','ETF 5D','ETF 20D','ETF 30D','DECYZJA','DCA1','DCA2','DCA3','BLOKADY'];
  const assets = (data.assets || []).map(a => [
    a.symbol, a.data_quality, translateReady_(a.readiness_status), a.zone_timeframes,
    a.trend_score_0_4, a.fomo_score_0_10, translateGeneric_(a.rotation_label), translateMarketPhase_(a.market_phase),
    a.long_flow_score_0_10, translateFlow_(a.long_flow_label), a.direct_etf_5d_usdm, a.direct_etf_20d_usdm, a.direct_etf_30d_usdm,

}

function writeLabPanel_(sheet, lab) {
  const rows = [['LAB — ANALIZA AKTYWA','','','','','','','','']];
  rows.push(['AKTYWO','TYP','TRYB','STAN DANYCH','WYNIK /10','OCENA','1H','4H','1D / GŁÓWNE INFORMACJE']);
  (lab.reports || []).forEach(a => {
    const pl = a.opis_po_polsku || {};
    const fast = a.fast_analysis || {};
    const t = a.technical || {};
    const h1=t['1H']||{}, h4=t['4H']||{}, d1=t['1D']||{};
    let details='Cena '+fmtCell_(d1.price)+' | RSI '+fmtCell_(d1.rsi14)+' | MFI '+fmtCell_(d1.mfi14)+' | FOMO '+fmtCell_(d1.fomo_0_10);
    if (a.symbol === 'BTC' || a.symbol === 'ETH') {
      const x = a.symbol === 'BTC' ? (a.btc_etf_context||{}) : (a.eth_etf_context||{});
      details += ' | ETF: '+(x.label_pl || 'BRAK DANYCH')+' | 5D '+fmtCell_(x.flow_5d_usdm)+' mln USD | 20D '+fmtCell_(x.flow_20d_usdm)+' mln USD';
    }
    rows.push([
      a.symbol || '', pl.typ_aktywa || a.asset_type || '', pl.tryb || a.mode || '', pl.stan_danych || '',
      fast.score_0_10 == null ? '' : fast.score_0_10, pl.wniosek_szybki || '',

    ]);
  });
  sheet.clearContents();
  sheet.getRange(1,1,rows.length,9).setValues(rows);
  sheet.getRange(1,1,1,9).merge().setFontWeight('bold').setHorizontalAlignment('center');
  sheet.getRange(2,1,1,9).setFontWeight('bold');

  [90,130,190,240,90,220,220,220,430].forEach((w,idx)=>sheet.setColumnWidth(idx+1,w));
}

function labTfText_(x) {
  if (!x || x.status !== 'OK') return 'BRAK PEŁNYCH DANYCH';
  return 'TREND '+fmtCell_(x.trend_0_4)+'/4 | RSI '+fmtCell_(x.rsi14)+' | MFI '+fmtCell_(x.mfi14)+' | FOMO '+fmtCell_(x.fomo_0_10)+'/10';
}

function fmtCell_(v) {
  if (v === null || v === undefined || v === '') return 'brak danych';
  const n = Number(v); return Number.isFinite(n) ? Math.round(n*100)/100 : String(v);
}
function valueOrBlank_(v) { const n=Number(v); return Number.isFinite(n) ? n : ''; }

function etfMeaning_(five, twenty, accel, fallback) {
  five=Number(five); twenty=Number(twenty); accel=Number(accel);
  if (Number.isFinite(five) && Number.isFinite(twenty)) {
    if (five>0 && twenty>0) return Number.isFinite(accel) && accel>0 ? 'NAPŁYW PRZYSPIESZA — zainteresowanie rośnie' : 'NAPŁYW KAPITAŁU — zainteresowanie dodatnie';
    if (five<0 && twenty<0) return Number.isFinite(accel) && accel<0 ? 'ODPŁYW PRZYSPIESZA — zainteresowanie słabnie' : 'ODPŁYW KAPITAŁU — zainteresowanie słabnie';
    return 'PRZEPŁYWY MIESZANE — brak jednolitego kierunku';
  }
  return fallback ? translateEtfTrend_(fallback) : 'ZA MAŁO DANYCH';
}

function globalEtfMeaning_(i) {
  const b5=Number(i.btc_etf_5d_usdm), b20=Number(i.btc_etf_20d_usdm), e5=Number(i.eth_etf_5d_usdm), e20=Number(i.eth_etf_20d_usdm);
  if ([b5,b20,e5,e20].every(Number.isFinite)) {
    if (b5>0 && b20>0 && e5>0 && e20>0) return 'KAPITAŁ NAPŁYWA RÓWNOCZEŚNIE DO BTC I ETH — szerokie zainteresowanie instytucjonalne krypto rośnie.';
    if (b5>0 && b20>0 && (e5<=0 || e20<=0)) return 'KAPITAŁ KONCENTRUJE SIĘ GŁÓWNIE NA BTC — zainteresowanie ETH nie potwierdza jeszcze szerokiego napływu.';
    if (e5>0 && e20>0 && (b5<=0 || b20<=0)) return 'ETH przyciąga kapitał mimo słabszego obrazu BTC — obserwuj możliwe przesunięcie zainteresowania.';
    if (b5<0 && b20<0 && e5<0 && e20<0) return 'KAPITAŁ ODPŁYWA RÓWNOCZEŚNIE Z BTC I ETH — zainteresowanie instytucjonalne słabnie.';
    return 'BTC i ETH pokazują różne kierunki przepływów — obraz jest mieszany.';
  }
  return 'Za mało kompletnych danych, aby ocenić wspólny obraz BTC i ETH.';
}

function translateEtfTrend_(v) {
  const m={'INFLOW_ACCELERATING':'NAPŁYW PRZYSPIESZA','INFLOW_POSITIVE':'NAPŁYW KAPITAŁU','OUTFLOW_ACCELERATING':'ODPŁYW PRZYSPIESZA','OUTFLOW_NEGATIVE':'ODPŁYW KAPITAŁU','MIXED':'PRZEPŁYWY MIESZANE','NEUTRAL':'NEUTRALNIE'};
  return m[String(v||'').toUpperCase()] || translateGeneric_(v);
}

function setupSettings_(sheet) {
  const defaults = [
    ['TRYB','RESEARCH'],['AUTO_EXECUTION','OFF'],['V7_TOUCH','OFF'],['LONG_TACTICAL_CAPITAL_SHARED','NO'],
    ['ALARMY_EKSTREMOW','ON'],
    ['RSI_GORNY','75'],['RSI_DOLNY','25'],['FOMO_TWARDY','8'],
    ['MFI_GORNY_OSTRZEZENIE','80'],['MFI_GORNY_SKRAJNY','90'],
    ['MFI_DOLNY_OSTRZEZENIE','20'],['MFI_DOLNY_SKRAJNY','10'],
    ['EMAIL_ALERTY','OFF'],['EMAIL_DO',''],['LAST_REFRESH','']
  ];
  if (sheet.getLastRow() === 0) {
    sheet.getRange(1,1,1,2).setValues([['PARAMETR','WARTOŚĆ']]);
    sheet.getRange(2,1,defaults.length,2).setValues(defaults);
  } else {
    const existing = {};
    sheet.getRange(1,1,sheet.getLastRow(),2).getValues().forEach(r => existing[String(r[0])] = true);
    defaults.forEach(r => { if (!existing[r[0]]) sheet.appendRow(r); });
  }
  sheet.setFrozenRows(1);
}

function setupExtremes_(sheet) {
  if (sheet.getLastRow() === 0) {
    sheet.getRange(1,1,1,11).setValues([['AKTYWNY','AKTYWO','INTERWAŁ','TYP','RSI','MFI','MACD %','MACD Z','FOMO','WAGA','CZAS DANYCH']]);
    sheet.setFrozenRows(1);
  }
}

function getSettingMap_(ss) {
  const sh = ss.getSheetByName(V8_SHEETS.settings); const out = {};
  if (!sh || sh.getLastRow() < 2) return out;
  sh.getRange(2,1,sh.getLastRow()-1,2).getValues().forEach(r => out[String(r[0]).trim()] = r[1]);
  return out;
}

function getExtremeSettings_(ss) {
  const s = getSettingMap_(ss);
  return {
    enabled: String(s.ALARMY_EKSTREMOW || 'ON').toUpperCase() === 'ON',
    rsiHigh: Number(s.RSI_GORNY || 75),
    rsiLow: Number(s.RSI_DOLNY || 25),
    fomoHard: Number(s.FOMO_TWARDY || 8),
    mfiWarnHigh: Number(s.MFI_GORNY_OSTRZEZENIE || 80),
    mfiExtremeHigh: Number(s.MFI_GORNY_SKRAJNY || 90),
    mfiWarnLow: Number(s.MFI_DOLNY_OSTRZEZENIE || 20),
    mfiExtremeLow: Number(s.MFI_DOLNY_SKRAJNY || 10),
    emailOn: String(s.EMAIL_ALERTY || 'OFF').toUpperCase() === 'ON',
    emailTo: String(s.EMAIL_DO || '').trim()
  };
}

function buildExtremeAlarms_(ss, tacticalJson) {
  const cfg = getExtremeSettings_(ss); if (!cfg.enabled) return [];
  const alarms = [];
  (tacticalJson.assets || []).forEach(a => {
    const t = a.timeframes || {};
    ['1H','4H','1D'].forEach(tf => {
      const x = t[tf] || {};
      const rsi = Number(x.rsi14), mfi = Number(x.mfi14), fomo = Number(x.fomo_score_0_10);
      const macdPct = Number(x.macd_hist_percentile), macdZ = Number(x.macd_hist_zscore);
      const ts = x.last_close_time_utc || '';
      if (Number.isFinite(rsi) && rsi >= cfg.rsiHigh) alarms.push(makeAlarm_(a.symbol,tf,'RSI WYKUPIENIE',rsi,mfi,macdPct,macdZ,fomo,rsi >= 80 ? 'EKSTREMALNY' : 'WYSOKI',ts));
      if (Number.isFinite(rsi) && rsi <= cfg.rsiLow) alarms.push(makeAlarm_(a.symbol,tf,'RSI WYPRZEDANIE',rsi,mfi,macdPct,macdZ,fomo,rsi <= 20 ? 'EKSTREMALNY' : 'WYSOKI',ts));
      if (Number.isFinite(mfi)) {
        if (mfi >= cfg.mfiExtremeHigh) alarms.push(makeAlarm_(a.symbol,tf,'MFI SKRAJNE WYKUPIENIE',rsi,mfi,macdPct,macdZ,fomo,'EKSTREMALNY',ts));
        else if (mfi >= cfg.mfiWarnHigh) alarms.push(makeAlarm_(a.symbol,tf,'MFI WYKUPIENIE — OSTRZEŻENIE',rsi,mfi,macdPct,macdZ,fomo,'WYSOKI',ts));
        if (mfi <= cfg.mfiExtremeLow) alarms.push(makeAlarm_(a.symbol,tf,'MFI SKRAJNE WYPRZEDANIE',rsi,mfi,macdPct,macdZ,fomo,'EKSTREMALNY',ts));
        else if (mfi <= cfg.mfiWarnLow) alarms.push(makeAlarm_(a.symbol,tf,'MFI WYPRZEDANIE — OSTRZEŻENIE',rsi,mfi,macdPct,macdZ,fomo,'WYSOKI',ts));
      }
      const ml = String(x.macd_extreme_label || 'NORMAL');
      if (ml === 'EXTREME_POSITIVE') alarms.push(makeAlarm_(a.symbol,tf,'MACD SKRAJNIE DODATNIE',rsi,mfi,macdPct,macdZ,fomo,'EKSTREMALNY',ts));
      else if (ml === 'EXTREME_NEGATIVE') alarms.push(makeAlarm_(a.symbol,tf,'MACD SKRAJNIE UJEMNE',rsi,mfi,macdPct,macdZ,fomo,'EKSTREMALNY',ts));
      else if (ml === 'WARNING_POSITIVE') alarms.push(makeAlarm_(a.symbol,tf,'MACD MOCNO DODATNIE — OSTRZEŻENIE',rsi,mfi,macdPct,macdZ,fomo,'WYSOKI',ts));
      else if (ml === 'WARNING_NEGATIVE') alarms.push(makeAlarm_(a.symbol,tf,'MACD MOCNO UJEMNE — OSTRZEŻENIE',rsi,mfi,macdPct,macdZ,fomo,'WYSOKI',ts));
      if (Number.isFinite(fomo) && fomo >= cfg.fomoHard) alarms.push(makeAlarm_(a.symbol,tf,'FOMO HARD BLOCK',rsi,mfi,macdPct,macdZ,fomo,fomo >= 9 ? 'EKSTREMALNY' : 'WYSOKI',ts));
    });
  });
  return alarms;
}

function makeAlarm_(symbol, tf, type, rsi, mfi, macdPct, macdZ, fomo, severity, ts) {
  return {aktywny:true,symbol:symbol,tf:tf,type:type,rsi:Number.isFinite(rsi)?rsi:'',mfi:Number.isFinite(mfi)?mfi:'',macdPct:Number.isFinite(macdPct)?macdPct:'',macdZ:Number.isFinite(macdZ)?macdZ:'',fomo:Number.isFinite(fomo)?fomo:'',severity:severity,ts:ts,key:[symbol,tf,type].join('|')};
}

function extremeLabel_(x, cfg) {
  const rsi = Number(x.rsi14), mfi = Number(x.mfi14), fomo = Number(x.fomo_score_0_10), labels = [];
  if (Number.isFinite(rsi) && rsi >= cfg.rsiHigh) labels.push('RSI WYKUPIENIE');
  if (Number.isFinite(rsi) && rsi <= cfg.rsiLow) labels.push('RSI WYPRZEDANIE');
  if (Number.isFinite(mfi)) {
    if (mfi >= cfg.mfiExtremeHigh) labels.push('MFI SKRAJNE WYKUPIENIE');
    else if (mfi >= cfg.mfiWarnHigh) labels.push('MFI WYKUPIENIE');
    if (mfi <= cfg.mfiExtremeLow) labels.push('MFI SKRAJNE WYPRZEDANIE');
    else if (mfi <= cfg.mfiWarnLow) labels.push('MFI WYPRZEDANIE');
  }
  const ml = String(x.macd_extreme_label || 'NORMAL');
  if (ml === 'EXTREME_POSITIVE') labels.push('MACD SKRAJNIE +');
  else if (ml === 'EXTREME_NEGATIVE') labels.push('MACD SKRAJNIE -');
  else if (ml === 'WARNING_POSITIVE') labels.push('MACD MOCNO +');
  else if (ml === 'WARNING_NEGATIVE') labels.push('MACD MOCNO -');
  if (Number.isFinite(fomo) && fomo >= cfg.fomoHard) labels.push('FOMO HARD');
  return labels.length ? labels.join(' + ') : 'BRAK';
}

function writeExtremeAlarms_(sheet, alarms) {
  sheet.clearContents();
  sheet.getRange(1,1,1,11).setValues([['AKTYWNY','AKTYWO','INTERWAŁ','TYP','RSI','MFI','MACD %','MACD Z','FOMO','WAGA','CZAS DANYCH']]);
  const rows = alarms.map(a => ['TAK',a.symbol,a.tf,a.type,a.rsi,a.mfi,a.macdPct,a.macdZ,a.fomo,a.severity,a.ts]);
  if (rows.length) sheet.getRange(2,1,rows.length,11).setValues(rows);
  else sheet.getRange(2,1,1,11).setValues([['NIE','','','BRAK AKTYWNYCH EKSTREMÓW','','','','','','','']]);
  sheet.setFrozenRows(1); sheet.autoResizeColumns(1,11);
}

function readActiveExtremeKeys_(sheet) {
  const out = {}; if (!sheet || sheet.getLastRow() < 2) return out;
  const rows = sheet.getRange(2,1,sheet.getLastRow()-1,Math.min(11,sheet.getLastColumn())).getValues();
  rows.forEach(r => { if (String(r[0]).toUpperCase() === 'TAK') out[[r[1],r[2],r[3]].join('|')] = true; });
  return out;
}

function appendExtremeHistory_(ss, oldKeys, alarms) {
  const sh = ss.getSheetByName(V8_SHEETS.history);
  if (sh.getLastRow() === 0) { sh.appendRow(['CZAS','AKTYWO','LONG','TACTICAL','RYZYKO','ZGODNOŚĆ','ALERT','POWÓD']); sh.setFrozenRows(1); }
  const now = new Date();
  alarms.forEach(a => { if (!oldKeys[a.key]) sh.appendRow([now,a.symbol,'','',a.severity,'','EKSTREMUM '+a.tf,a.type+' | RSI='+a.rsi+' | MFI='+a.mfi+' | MACD%='+a.macdPct+' | MACDz='+a.macdZ+' | FOMO='+a.fomo]); });
}

function notifyNewExtremeAlarms_(ss, oldKeys, alarms) {
  const fresh = alarms.filter(a => !oldKeys[a.key]); if (!fresh.length) return;
  const text = fresh.map(a => a.symbol+' '+a.tf+' — '+a.type+' (RSI '+a.rsi+', MFI '+a.mfi+', MACD% '+a.macdPct+', MACDz '+a.macdZ+', FOMO '+a.fomo+')').join('\n');
  ss.toast(text.slice(0,250), 'NOWE EKSTREMA V8', 8);
  const cfg = getExtremeSettings_(ss);
  if (cfg.emailOn && cfg.emailTo) MailApp.sendEmail(cfg.emailTo, 'V8 DUAL ENGINE — alarm ekstremum', text);
}

function stampRefresh_(ss) {
  const sh = ss.getSheetByName(V8_SHEETS.settings);
  const vals = sh.getRange(1,1,Math.max(1,sh.getLastRow()),2).getValues();
  for (let i=0; i<vals.length; i++) if (vals[i][0] === 'LAST_REFRESH') { sh.getRange(i+1,2).setValue(new Date()); return; }
}

function readPanelState_(sheet) {
  const out = {}; if (!sheet || sheet.getLastRow() < 2) return out;
  const values = sheet.getDataRange().getValues(); const head = values[0].map(String); const idx = {}; head.forEach((h,i)=>idx[h]=i);
  values.slice(1).forEach(r => {
    const sym = String(r[idx['AKTYWO']] || '').trim(); if (!sym) return;
    out[sym] = {LONG:r[idx['LONG']], TACTICAL:r[idx['TACTICAL']], RYZYKO:r[idx['RYZYKO 0–10']], ZGODNOSC:r[idx['ZGODNOŚĆ']], ALERT:r[idx['ALERT']]};
  });
  return out;
}

function appendHistoryOnChanges_(ss, oldState, newState) {
  const sh = ss.getSheetByName(V8_SHEETS.history);
  if (sh.getLastRow() === 0) { sh.appendRow(['CZAS','AKTYWO','LONG','TACTICAL','RYZYKO','ZGODNOŚĆ','ALERT','POWÓD']); sh.setFrozenRows(1); }
  const now = new Date();
  Object.keys(newState).forEach(sym => {
    const n = newState[sym], o = oldState[sym];
    const changed = !o || ['LONG','TACTICAL','RYZYKO','ZGODNOSC','ALERT'].some(k => String(o[k]) !== String(n[k]));
    if (changed) sh.appendRow([now,sym,n.LONG,n.TACTICAL,n.RYZYKO,n.ZGODNOSC,n.ALERT,o ? 'ZMIANA STATUSU' : 'PIERWSZY ZAPIS']);
  });
}

function formatPanel_(sheet) {
  if (!sheet) return;
  sheet.setFrozenRows(1); sheet.setFrozenColumns(1); sheet.getRange('A:G').setVerticalAlignment('middle');
  [90,100,150,190,100,140,190].forEach((w,i)=>sheet.setColumnWidth(i+1,w));
  if (sheet.getLastRow() >= 2) sheet.getRange(2,1,sheet.getLastRow()-1,7).setWrap(true);
}
